async function stage(page, name) {
  const info = await page.evaluate((patSrc) => {
    const re = new RegExp(patSrc, "i");
    const hits = [];
    const w = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
    let n;
    while ((n = w.nextNode())) {
      const t = (n.nodeValue || "").trim();
      if (t.length < 2 || t.length > 90 || !re.test(t)) continue;
      const el = n.parentElement; if (!el) continue;
      const r = el.getBoundingClientRect();
      const cx = r.x + r.width / 2, cy = r.y + r.height / 2;
      const hit = (r.width > 0 && r.height > 0) ? document.elementFromPoint(cx, cy) : null;
      hits.push({ text: t.slice(0, 60), tag: el.tagName.toLowerCase(),
                  cls: (el.className || "").toString().slice(0, 44),
                  box: { x: Math.round(r.x), y: Math.round(r.y), w: Math.round(r.width), h: Math.round(r.height) },
                  centre: { cx: Math.round(cx), cy: Math.round(cy) },
                  in_viewport: r.top >= 0 && r.bottom <= window.innerHeight,
                  hit_is_self: hit ? (hit === el || el.contains(hit) || hit.contains(el)) : false,
                  clickable_ancestor: !!el.closest("button,[role=button],a,input,summary") });
    }
    return { url: location.pathname + location.search, chars: (document.body.innerText || "").length,
             text_hits: hits.slice(0, 25),
             buttons: Array.from(document.querySelectorAll("button,[role=button],summary"))
