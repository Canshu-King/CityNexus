def http(url, data=None, headers=None, timeout=60):
    req = urllib.request.Request(url, data=data, method="POST" if data is not None else "GET",
                                 headers=headers or {})
    try:
        with urllib.request.urlopen(req, timeout=timeout) as r:
            return r.status, r.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()
    except Exception as e:  # noqa: BLE001
        return None, (type(e).__name__ + str(e))[:150].encode()


def free_port(wanted, taken=()):
    p = wanted
    while p < wanted + 80:
        if p not in FORBIDDEN and p not in taken:
            with socket.socket() as s:
                try:
                    s.bind(("127.0.0.1", p))
                    return p
                except OSError:
                    pass
        p += 1
    raise SystemExit("找不到空闲私有端口")

class H(BaseHTTPRequestHandler):
    def log_message(self, *a):
        pass

    def _send(self, code, body, ctype):
        self.send_response(code)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        try:
            self.wfile.write(body)
        except (ConnectionAbortedError, ConnectionResetError, BrokenPipeError):
            pass

    def do_GET(self):
        p = self.path.split("?")[0]
        if p.startswith("/api/"):
            st, body = http("http://127.0.0.1:%d%s" % (api, self.path))
            self._send(st or 502, body, "application/json")
            return
        f = (DIST / (p.lstrip("/") or "index.html")).resolve()
        ctype = ("text/javascript" if f.suffix == ".js" else "text/css" if f.suffix == ".css"
                 else "text/html" if f.suffix == ".html" else "application/octet-stream")
        if not str(f).startswith(str(DIST)) or not f.is_file():
            f, ctype = DIST / "index.html", "text/html"
        self._send(200, f.read_bytes(), ctype)

    do_POST = do_GET
