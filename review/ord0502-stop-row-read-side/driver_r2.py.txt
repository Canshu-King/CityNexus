# -*- coding: utf-8 -*-
"""读侧那半问的第二把尺（r1 已冻结不改）：
① 哪条路由能读到站点行 —— 站点行的姓名字段在这代叫 `name` 不叫 `title`（r1 按 `title` 判，
   于是把自己的读数误报成"不成立"；这是仪器的错，先记下来）；
② 行里哪个字段指回行程 —— 站点行自身带不带回指、回指在哪一跳上；
③ 把「读站点行的自由函数」往上接到**真正调用它的路由**（源级清单，不靠猜）。
所有 HTTP 与库表读数都取自现读尖的导出与新库，一次进程一趟。"""
from __future__ import annotations

import importlib.util
import json
import os
import re
import secrets
import sqlite3
import subprocess
import sys
import time
import urllib.error
import urllib.request
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
RUNNER = WT2 / "review/ord0502/run_ord0502_stack_r1.py"
ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
MAIN = "backend/app/main.py"
NEEDLES = ("select(Place)", "Place.id", "Place.city_code", "session.get(Place")
BAD: list[str] = []


def g(*a: str) -> str:
    r = subprocess.run(["git", *a], cwd=str(WT2), capture_output=True, text=True,
                       encoding="utf-8", errors="replace")
    if r.returncode != 0:
        BAD.append("GIT_FAIL %s rc=%d" % (" ".join(a)[:60], r.returncode))
    return r.stdout or ""


def top_level_defs(lines):
    """返回 [(def_line0, name, end_line0)]：顶层 def/async def 的边界。"""

    starts = [i for i, ln in enumerate(lines) if re.match(r"^(async )?def ", ln)]
    out = []
    for k, i in enumerate(starts):
        nxt = starts[k + 1] if k + 1 < len(starts) else len(lines)
        # 顶层非缩进、非装饰器、非空的行也可能截断函数体（如模块级赋值）
        j = i + 1
        while j < nxt:
            ln = lines[j]
            if ln.strip() and not ln.startswith((" ", "\t")) and not ln.startswith("@"):
                nxt = j
                break
            j += 1
        out.append((i, re.sub(r"^(async )?def (\w+).*", r"\2", lines[i]), nxt))
    return out


def route_of_def(lines, i, defs):
    for d in lines[max(0, i - 6):i][::-1]:
        m = re.match(r'^@app\.(get|post|patch|put|delete)\(\s*"([^"]+)"', d)
        if m:
            return m.group(1).upper(), m.group(2)
    return None


def main() -> int:
    tip = g("ls-remote", "--refs", "origin", MIRROR).split("\t")[0].strip()
    if len(tip) != 40:
        print("REFUSED: 读不到产品线尖")
        return 1
    lines = g("show", "%s:%s" % (tip, MAIN)).split("\n")
    defs = top_level_defs(lines)
    by_name = {n: (i, e) for i, n, e in defs}

    readers = {}
    for i, name, end in defs:
        body = "\n".join(lines[i:end])
        found = {nd: body.count(nd) for nd in NEEDLES if nd in body}
        if found:
            readers[name] = {"def_line": i + 1, "kind": "route" if route_of_def(lines, i, defs)
                             else "free_function", "route": route_of_def(lines, i, defs),
                             "place_reads": found}
    # 自由函数 → 调用它的路由（一层，够回答"谁能读到"）
    callers = {}
    for fn in readers:
        if readers[fn]["kind"] != "free_function":
            continue
        hits = []
        for i, name, end in defs:
            if name == fn:
                continue
            body = "\n".join(lines[i:end])
            if re.search(r"\b%s\(" % re.escape(fn), body):
                r = route_of_def(lines, i, defs)
                hits.append({"in_function": name, "route": r, "def_line": i + 1})
        callers[fn] = hits
    out = {"line_tip": tip, "main_py_blob": g("rev-parse", "%s:%s" % (tip, MAIN)).strip(),
           "place_readers_at_source": readers, "free_function_callers": callers,
           "place_readers_count": len(readers),
           "routes_directly_reading_place": sorted({tuple(readers[k]["route"]) for k in readers
                                                    if readers[k]["route"]})}

    spec = importlib.util.spec_from_file_location("ord0502_runner", str(RUNNER))
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    run_id = "READSIDE2-" + time.strftime("%H%M%S")
    snap = Path("D:/moodle") / ("_qoder_readside2_%s" % run_id)
    snap.mkdir(parents=True, exist_ok=True)
    proc = subprocess.run(["git", "archive", "--format=tar", tip], cwd=str(WT2),
                          capture_output=True)
    if proc.returncode != 0:
        print("REFUSED: archive rc=%d" % proc.returncode)
        return 1
    tar = snap / "a.tar"
    tar.write_bytes(proc.stdout)
    import tarfile
    with tarfile.open(str(tar)) as t:
        t.extractall(path=str(snap))
    tar.unlink()
    if g("hash-object", "--no-filters", str(snap / MAIN)).strip() != out["main_py_blob"]:
        print("REFUSED: 导出的 main.py 与尖上那枚字节不等")
        return 1

    db = snap / "backend" / ("%s.db" % run_id)
    port = mod.free_port(int(os.environ.get("READSIDE_PORT", "4809")), "backend")
    env = mod.env_base(run_id, db, port)
    env["CITYNEXUS_DEMO_MODE"] = "true"
    token = secrets.token_hex(16)
    env["CITYNEXUS_DEMO_SCENE_TOKEN"] = token
    base = "http://127.0.0.1:%d" % port
    hdr = {"X-Demo-Scene-Token": token, "Idempotency-Key": "rs2-" + run_id,
           "Content-Type": "application/json"}
    out.update({"run_id": run_id, "backend_port": port, "db": db.as_posix(),
                "token_names_only": ["CITYNEXUS_DEMO_SCENE_TOKEN", "X-Demo-Scene-Token"]})

    def get(path):
        try:
            with urllib.request.urlopen(base + path, timeout=90) as r:
                return r.status, r.read(), dict(r.headers).get("Content-Type")
        except urllib.error.HTTPError as e:
            return e.code, e.read(), dict(e.headers).get("Content-Type")
        except Exception as e:  # noqa: BLE001
            return None, (type(e).__name__ + str(e))[:120].encode(), None

    child = subprocess.Popen([mod.PY, "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1",
                              "--port", str(port)], cwd=str(snap / "backend"), env=env,
                             stdout=(ART / ("uvicorn_%s.log" % run_id)).open("w", encoding="utf-8"),
                             stderr=subprocess.STDOUT)
    out["child_pid"] = child.pid
    try:
        ok, _ = mod.wait_http(base + "/api/v1/health")
        hs, hb, _ = get("/api/v1/health")
        out["health"] = {"ok": ok, "http_status": hs,
                         "carries_this_run_id": run_id in hb.decode("utf-8", "replace")}
        if not out["health"]["carries_this_run_id"]:
            BAD.append("归属未由本 run id 证")
        r = urllib.request.Request(base + "/api/v1/demo/scene/ensure", data=b"{}", method="POST",
                                   headers=hdr)
        with urllib.request.urlopen(r, timeout=120) as resp:
            rec = json.loads(resp.read().decode("utf-8"))
        jid = rec.get("journey_id")
        out["ensure"] = {"http_status": resp.status, "state": rec.get("state"),
                         "journey_id": jid, "journey_status": rec.get("journey_status")}
        if resp.status != 200 or rec.get("state") != "READY":
            BAD.append("新尖上新库首供不是 200 READY")
        con = sqlite3.connect("file:%s?mode=ro" % db.as_posix(), uri=True)
        pv = str(con.execute("SELECT active_plan_version_id FROM journeys WHERE id=?",
                             (jid,)).fetchone()[0])
        pid = con.execute("SELECT place_id FROM activities WHERE plan_version_id=? "
                          "AND place_id IS NOT NULL ORDER BY sequence_no LIMIT 1",
                          (pv,)).fetchone()[0]
        out["db_shapes"] = {}
        for t in ("places", "activities", "plan_versions", "journey_legs"):
            cols = [x[1] for x in con.execute('PRAGMA table_info("%s")' % t)]
            out["db_shapes"][t] = {
                "columns": cols, "has_journey_id": "journey_id" in cols,
                "has_plan_version_id": "plan_version_id" in cols, "has_place_id": "place_id" in cols,
                "rows": con.execute('SELECT count(*) FROM "%s"' % t).fetchone()[0]}
        out["db_shapes"]["activities"]["rows_on_active_plan_version"] = con.execute(
            "SELECT count(*) FROM activities WHERE plan_version_id=?", (pv,)).fetchone()[0]
        out["db_shapes"]["places"]["rows_in_this_city"] = con.execute(
            "SELECT count(*) FROM places WHERE city_code='CNX-DEMO'").fetchone()[0]
        con.close()

        probes = ["/api/v1/places?city_code=CNX-DEMO", "/api/v1/places/%s" % pid,
                  "/api/v1/journeys/%s" % jid, "/api/v1/journeys/%s/plans" % jid,
                  "/api/v1/journeys/%s/calendar.ics" % jid, "/api/v1/demo/scene"]
        res = []
        for p in probes:
            st, body, ctype = get(p)
            j = None
            try:
                j = json.loads(body.decode("utf-8"))
            except Exception:  # noqa: BLE001
                j = None
            rec2 = {"path": p.replace(jid, "<jid>").replace(pid, "<place_id>"),
                    "http_status": st, "bytes": len(body), "content_type": ctype}
            if isinstance(j, list):
                rec2["top_level"] = "array"
                rec2["length"] = len(j)
                rec2["first_item_keys"] = sorted(j[0].keys()) if j and isinstance(j[0], dict) else []
                bag = {}
                stack = [j]
                while stack:
                    cur = stack.pop()
                    if isinstance(cur, dict):
                        for k, v in cur.items():
                            bag[k] = bag.get(k, 0) + 1
                            stack.append(v)
                    elif isinstance(cur, list):
                        stack.extend(cur)
                rec2["field_occurrences"] = {k: bag.get(k, 0) for k in
                                             ("id", "name", "title", "place_id", "journey_id",
                                              "plan_version_id", "activities", "city_code")}
            elif isinstance(j, dict):
                rec2["top_level"] = "object"
                rec2["first_item_keys"] = sorted(j.keys())[:16]
            else:
                rec2["head"] = body[:70].decode("utf-8", "replace")
            res.append(rec2)
        out["probes"] = res
    finally:
        child.terminate()
        try:
            child.wait(timeout=60)
        except subprocess.TimeoutExpired:
            child.kill()
        out["child_exit"] = child.poll()

    pl = next((r for r in out["probes"] if r["path"].startswith("/api/v1/places?")), None)
    if not pl or pl["http_status"] != 200:
        BAD.append("站点目录路由没答 200")
    elif pl.get("field_occurrences", {}).get("name", 0) < 1 or pl.get("length", 0) < 1:
        BAD.append("站点路由的答复里没有 name 字段或长度为零")
    byid = next((r for r in out["probes"] if r["path"] == "/api/v1/places/<place_id>"), None)
    out["place_by_id_route_exists"] = bool(byid and byid["http_status"] == 200)
    if out["db_shapes"]["places"]["has_journey_id"]:
        BAD.append("places 这次带了 journey_id 列")
    if not out["db_shapes"]["activities"]["has_plan_version_id"]:
        BAD.append("activities 没有 plan_version_id 列")
    if not any(k == "search_places" for k in out["place_readers_at_source"]):
        BAD.append("源级清单里没有 search_places")
    out["BAD"] = BAD
    dst = ART / ("place_read_side_r2_%s.json" % run_id)
    dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + "\n", encoding="utf-8",
                   newline=chr(10))
    brief = {k: out[k] for k in ("line_tip", "run_id", "backend_port", "ensure",
                                 "place_readers_count", "routes_directly_reading_place",
                                 "place_by_id_route_exists", "db_shapes", "probes")}
    print(json.dumps(brief, ensure_ascii=False, indent=2)[:3800])
    print("自由函数→调用它们的路由：")
    for fn, hits in out["free_function_callers"].items():
        named = [(h["route"] or ("(无路由装饰器)", h["in_function"]))[0:2] for h in hits]
        print("  %s @ %s → %s" % (fn, out["place_readers_at_source"][fn]["def_line"], named[:4]))
    print("ARTIFACT", dst, "| BAD", BAD)
    return 1 if BAD else 0


if __name__ == "__main__":
    sys.exit(main())
