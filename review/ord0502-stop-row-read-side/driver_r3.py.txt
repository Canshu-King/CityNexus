# -*- coding: utf-8 -*-
"""读侧第三把尺（r1、r2 都冻住不改，因为它们各自的错已被点名）：
r1 用 `title` 判站点行 → 把自己的读数误报成"不成立"（这代字段叫 `name`）；
r2 切函数体时从 `def` 下一行就开始找顶格行 → 多行签名的收尾 `) -> …:` 被当成体外第一枚顶格行，
   于是**签名之后整段函数体被切掉**，`search_places` 这类路由从源级清单里凭空消失。
r3 先把签名读到冒号结束，再取体；并且自由函数要往上接到真正调用它的路由。"""
from __future__ import annotations

import importlib.util
import json
import os
import re
import secrets
import sqlite3
import subprocess
import sys
import time
import urllib.error
import urllib.request
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
RUNNER = WT2 / "review/ord0502/run_ord0502_stack_r1.py"
ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
MAIN = "backend/app/main.py"
NEEDLES = ("select(Place)", "Place.id", "Place.city_code", "session.get(Place")
DEC = re.compile(r'^@app\.(get|post|patch|put|delete)\(\s*"([^"]+)"')
BAD: list[str] = []


def g(*a: str) -> str:
    r = subprocess.run(["git", *a], cwd=str(WT2), capture_output=True, text=True,
                       encoding="utf-8", errors="replace")
    if r.returncode != 0:
        BAD.append("GIT_FAIL %s rc=%d" % (" ".join(a)[:60], r.returncode))
    return r.stdout or ""


def sig_end(lines, i):
    k = i
    while k < len(lines) - 1 and not lines[k].rstrip().endswith(":"):
        k += 1
    return k


def route_before(lines, i):
    for k in range(max(0, i - 8), i):
        m = DEC.match(lines[k])
        if m:
            return m.group(1).upper(), m.group(2)
    return None


def funcs(lines):
    starts = [i for i, ln in enumerate(lines) if re.match(r"^(async )?def ", ln)]
    out = []
    for pos, i in enumerate(starts):
        nxt = starts[pos + 1] if pos + 1 < len(starts) else len(lines)
        se = sig_end(lines, i)
        j = se + 1
        while j < nxt:
            ln = lines[j]
            if ln.strip() and not ln.startswith((" ", "\t")):
                nxt = j
                break
            j += 1
        out.append({"name": re.sub(r"^(async )?def (\w+).*", r"\2", lines[i]), "def_line": i + 1,
                    "sig_end": se + 1, "body_from": se + 2, "body_to": nxt,
                    "route": route_before(lines, i)})
    return out


def main() -> int:
    tip = g("ls-remote", "--refs", "origin", MIRROR).split("\t")[0].strip()
    if len(tip) != 40:
        print("REFUSED: 读不到产品线尖")
        return 1
    lines = g("show", "%s:%s" % (tip, MAIN)).split("\n")
    if len(lines) < 1000:
        print("REFUSED: main.py 读不通")
        return 1
    F = funcs(lines)
    readers = {}
    for f in F:
        body = "\n".join(lines[f["sig_end"]:f["body_to"]])
        found = {n: body.count(n) for n in NEEDLES if n in body}
        if found:
            readers[f["name"]] = {"def_line": f["def_line"], "body_lines": f["body_to"] - f["sig_end"],
                                  "route": list(f["route"]) if f["route"] else None,
                                  "place_reads": found}
    by_name = {f["name"]: f for f in F}
    callers = {}
    for fn in readers:
        if readers[fn]["route"]:
            continue
        hits = []
        for f in F:
            if f["name"] == fn:
                continue
            body = "\n".join(lines[f["sig_end"]:f["body_to"]])
            if re.search(r"\b%s\(" % re.escape(fn), body):
                hits.append({"in_function": f["name"], "def_line": f["def_line"],
                             "route": list(f["route"]) if f["route"] else None})
        callers[fn] = hits
    if "search_places" not in readers:
        BAD.append("源级清单里没有 search_places —— 切法还错")
    direct_routes = sorted({tuple(v["route"]) for v in readers.values() if v["route"]})
    out = {"line_tip": tip, "main_py_blob": g("rev-parse", "%s:%s" % (tip, MAIN)).strip(),
           "functions_scanned": len(F), "place_reader_count": len(readers),
           "place_readers": readers, "free_function_callers": callers,
           "routes_reading_place_rows_directly": direct_routes}

    spec = importlib.util.spec_from_file_location("ord0502_runner", str(RUNNER))
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    run_id = "READSIDE3-" + time.strftime("%H%M%S")
    snap = Path("D:/moodle") / ("_qoder_readside3_%s" % run_id)
    snap.mkdir(parents=True, exist_ok=True)
    proc = subprocess.run(["git", "archive", "--format=tar", tip], cwd=str(WT2),
                          capture_output=True)
    if proc.returncode != 0:
        print("REFUSED: archive rc=%d" % proc.returncode)
        return 1
    tar = snap / "a.tar"
    tar.write_bytes(proc.stdout)
    import tarfile
    with tarfile.open(str(tar)) as t:
        t.extractall(path=str(snap))
    tar.unlink()
    if g("hash-object", "--no-filters", str(snap / MAIN)).strip() != out["main_py_blob"]:
        print("REFUSED: 导出的 main.py 与尖上那枚不等")
        return 1

    db = snap / "backend" / ("%s.db" % run_id)
    port = mod.free_port(int(os.environ.get("READSIDE_PORT", "4811")), "backend")
    env = mod.env_base(run_id, db, port)
    env["CITYNEXUS_DEMO_MODE"] = "true"
    token = secrets.token_hex(16)
    env["CITYNEXUS_DEMO_SCENE_TOKEN"] = token
    base = "http://127.0.0.1:%d" % port
    out.update({"run_id": run_id, "backend_port": port, "db": db.as_posix(),
                "token_names_only": ["CITYNEXUS_DEMO_SCENE_TOKEN", "X-Demo-Scene-Token"]})

    def get(path):
        try:
            with urllib.request.urlopen(base + path, timeout=90) as r:
                return r.status, r.read(), {k.lower(): v for k, v in dict(r.headers).items()}
        except urllib.error.HTTPError as e:
            return e.code, e.read(), {k.lower(): v for k, v in dict(e.headers).items()}
        except Exception as e:  # noqa: BLE001
            return None, (type(e).__name__ + str(e))[:120].encode(), {}

    child = subprocess.Popen([mod.PY, "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1",
                              "--port", str(port)], cwd=str(snap / "backend"), env=env,
                             stdout=(ART / ("uvicorn_%s.log" % run_id)).open("w", encoding="utf-8"),
                             stderr=subprocess.STDOUT)
    out["child_pid"] = child.pid
    try:
        ok, _ = mod.wait_http(base + "/api/v1/health")
        hs, hb, _ = get("/api/v1/health")
        out["health"] = {"ok": ok, "http_status": hs,
                         "carries_this_run_id": run_id in hb.decode("utf-8", "replace")}
        if not out["health"]["carries_this_run_id"]:
            BAD.append("归属未由本 run id 证")
        rq = urllib.request.Request(base + "/api/v1/demo/scene/ensure", data=b"{}", method="POST",
                                    headers={"X-Demo-Scene-Token": token,
                                             "Idempotency-Key": "rs3-" + run_id,
                                             "Content-Type": "application/json"})
        with urllib.request.urlopen(rq, timeout=120) as resp:
            rec = json.loads(resp.read().decode("utf-8"))
        jid = rec.get("journey_id")
        out["ensure"] = {"http_status": resp.status, "state": rec.get("state"),
                         "journey_id": jid, "journey_status": rec.get("journey_status")}
        if resp.status != 200 or rec.get("state") != "READY":
            BAD.append("现尖上新库首供不是 200 READY")
        con = sqlite3.connect("file:%s?mode=ro" % db.as_posix(), uri=True)
        pv = str(con.execute("SELECT active_plan_version_id FROM journeys WHERE id=?",
                             (jid,)).fetchone()[0])
        pid = con.execute("SELECT place_id FROM activities WHERE plan_version_id=? "
                          "AND place_id IS NOT NULL ORDER BY sequence_no LIMIT 1",
                          (pv,)).fetchone()[0]
        out["db_shapes"] = {}
        for t in ("places", "activities", "plan_versions", "journey_legs"):
            cols = [x[1] for x in con.execute('PRAGMA table_info("%s")' % t)]
            out["db_shapes"][t] = {"columns": cols, "has_journey_id": "journey_id" in cols,
                                   "has_plan_version_id": "plan_version_id" in cols,
                                   "has_place_id": "place_id" in cols,
                                   "rows": con.execute('SELECT count(*) FROM "%s"' % t).fetchone()[0]}
        out["db_shapes"]["activities"]["rows_on_active_plan_version"] = con.execute(
            "SELECT count(*) FROM activities WHERE plan_version_id=?", (pv,)).fetchone()[0]
        out["db_shapes"]["places"]["rows_in_this_city"] = con.execute(
            "SELECT count(*) FROM places WHERE city_code='CNX-DEMO'").fetchone()[0]
        con.close()

        def probe(path, label):
            st, body, hd = get(path)
            j = None
            try:
                j = json.loads(body.decode("utf-8"))
            except Exception:  # noqa: BLE001
                j = None
            rec2 = {"label": label, "path": path.replace(jid, "<jid>").replace(pid, "<place_id>"),
                    "http_status": st, "bytes": len(body), "content_type": hd.get("content-type")}
            stack, bag = [j], {}
            while stack:
                cur = stack.pop()
                if isinstance(cur, dict):
                    for k, v in cur.items():
                        bag[k] = bag.get(k, 0) + 1
                        stack.append(v)
                elif isinstance(cur, list):
                    bag["__array_items__"] = bag.get("__array_items__", 0) + len(cur)
                    stack.extend(cur)
            rec2["field_occurrences"] = {k: bag.get(k, 0) for k in
                                         ("id", "name", "title", "place_id", "journey_id",
                                          "plan_version_id", "activities", "city_code")}
            if isinstance(j, list):
                rec2["top_level"] = "array"
                rec2["length"] = len(j)
                rec2["first_item_keys"] = sorted(j[0].keys()) if j and isinstance(j[0], dict) else []
            elif isinstance(j, dict):
                rec2["top_level"] = "object"
                rec2["first_item_keys"] = sorted(j.keys())[:18]
            else:
                rec2["head"] = body[:80].decode("utf-8", "replace")
            return rec2

        out["probes"] = [
            probe("/api/v1/places?city_code=CNX-DEMO", "城市目录里的站点行"),
            probe("/api/v1/places/%s" % pid, "按身份取单枚站点"),
            probe("/api/v1/journeys/%s" % jid, "行程本体"),
            probe("/api/v1/journeys/%s/plans" % jid, "计划族（安排行与站点身份都在里面）"),
            probe("/api/v1/journeys/%s/calendar.ics" % jid, "日历导出"),
            probe("/api/v1/demo/scene", "演示舞台的门牌"),
        ]
    finally:
        child.terminate()
        try:
            child.wait(timeout=60)
        except subprocess.TimeoutExpired:
            child.kill()
        out["child_exit"] = child.poll()

    P = {p["label"]: p for p in out["probes"]}
    cat = P.get("城市目录里的站点行", {})
    if cat.get("http_status") != 200 or cat.get("length", 0) < 1:
        BAD.append("城市目录路由没答到站点行")
    elif not (set(cat.get("first_item_keys", [])) >= {"id", "name", "city_code"}):
        BAD.append("站点行的字段形状里没有 id/name/city_code")
    plans = P.get("计划族（安排行与站点身份都在里面）", {})
    if plans.get("http_status") != 200 or plans["field_occurrences"].get("place_id", 0) < 1:
        BAD.append("plans 路由里没有站点身份字段")
    if out["db_shapes"]["places"]["has_journey_id"]:
        BAD.append("places 这次带了 journey_id 列，块里的回指那句要改")
    if not (out["db_shapes"]["activities"]["has_plan_version_id"]
            and out["db_shapes"]["activities"]["has_place_id"]):
        BAD.append("activities 缺 plan_version_id 或 place_id 列")
    if not out["db_shapes"]["plan_versions"]["has_journey_id"]:
        BAD.append("plan_versions 没有 journey_id 列")
    out["place_by_id_route_exists"] = P.get("按身份取单枚站点", {}).get("http_status") == 200
    out["journey_only_route_carries_place_rows"] = P.get("行程本体", {}).get(
        "field_occurrences", {}).get("place_id", 0)
    out["BAD"] = BAD
    dst = ART / ("place_read_side_r3_%s.json" % run_id)
    dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + "\n", encoding="utf-8",
                   newline=chr(10))
    print(json.dumps({k: out[k] for k in ("line_tip", "run_id", "backend_port", "functions_scanned",
                                          "place_reader_count", "routes_reading_place_rows_directly",
                                          "ensure", "db_shapes", "probes",
                                          "place_by_id_route_exists")},
                     ensure_ascii=False, indent=2)[:4200])
    print("自由函数 → 调用它们的路由：")
    for fn, hits in out["free_function_callers"].items():
        rs = sorted({(h["route"] or [None, None])[1] for h in hits})
        print("  %s（def %s）被 %d 处调用｜路由：%s" % (fn, out["place_readers"][fn]["def_line"],
                                                     len(hits), rs[:6]))
    print("ARTIFACT", dst, "| BAD", BAD)
    return 1 if BAD else 0


if __name__ == "__main__":
    sys.exit(main())
