# -*- coding: utf-8 -*-
"""定案：`select(Place)` 这类语句到底落在哪枚函数的行界里。
两把我自己的尺给的集合不一致（一把把 `GET /api/v1/plans/{plan_version_id}` 算成直接读站点，
另一把算成 `POST …/select`），所以不再靠正则猜装饰器归属：
用 ast 取每枚顶层函数的 (def 行, end 行)，再用 `git grep -n` 取每一枚命中点的行号，
把命中点**按行号落进函数行界**——落进哪枚就是哪枚读的，装饰器只是那枚函数对外的门面。"""
from __future__ import annotations

import ast
import json
import re
import subprocess
import sys
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
MAIN = "backend/app/main.py"
NEEDLES = ("select(Place)", "Place.id", "Place.city_code", "session.get(Place")
DEC = re.compile(r'^@app\.(get|post|patch|put|delete)\(\s*(?:[a-z_]+=\([^()]*\),\s*)*"([^"]+)"')


def g(*a: str) -> str:
    return subprocess.run(["git", *a], cwd=str(WT2), capture_output=True, text=True,
                          encoding="utf-8", errors="replace").stdout


def attributions(tip: str):
    src = g("show", "%s:%s" % (tip, MAIN))
    lines = src.split("\n")
    tree = ast.parse(src)
    fns = []
    for node in tree.body:
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            decorators = []
            for d in node.decorator_list:
                if isinstance(d, ast.Call) and isinstance(d.func, ast.Attribute) \
                        and d.func.attr in ("get", "post", "patch", "put", "delete") \
                        and d.args and isinstance(d.args[0], ast.Constant) \
                        and isinstance(d.args[0].value, str):
                    decorators.append((d.func.attr.upper(), d.args[0].value, d.lineno))
            head = max([d[2] for d in decorators], default=node.lineno)
            fns.append({"name": node.name, "def_line": node.lineno, "head_line": head,
                        "end_line": node.end_lineno, "routes": decorators})
    fns.sort(key=lambda f: f["head_line"])
    # 命中点：用正则按行扫（比 git grep 稳，避免把 rev 前缀算进偏移）
    points = []
    for i, ln in enumerate(lines, 1):
        for n in NEEDLES:
            if n in ln:
                points.append({"line": i, "needle": n, "text": ln.strip()[:90]})
    owner_of = {}
    for pt in points:
        owner = None
        for f in fns:
            if f["def_line"] <= pt["line"] <= f["end_line"]:
                owner = f
                break
        key = owner["name"] if owner else "(不属于任何顶层函数)"
        owner_of.setdefault(key, {"function": owner["name"] if owner else key,
                                  "def_line": owner["def_line"] if owner else None,
                                  "end_line": owner["end_line"] if owner else None,
                                  "routes": [list(r[0:2]) for r in owner["routes"]] if owner else [],
                                  "hits": []})
        owner_of[key]["hits"].append(pt)
    return {"line_tip": tip,
            "main_py_blob": g("rev-parse", "%s:%s" % (tip, MAIN)).strip(),
            "top_level_functions": len(fns),
            "route_decorators_total": sum(len(f["routes"]) for f in fns),
            "place_hit_points": len(points),
            "owners": sorted(owner_of.values(), key=lambda o: (o["def_line"] or 0))}


def main() -> int:
    tips = [g("ls-remote", "--refs", "origin", MIRROR).split("\t")[0].strip()]
    src_tips = [l for l in g("log", "--format=%H", "-5", tips[0]).split("\n") if l.strip()]
    booked = "74f5d8df082532f20e66bdba2b84311043f023b5"
    run = {t: attributions(t) for t in {tips[0], booked} | set(src_tips[:3])}
    for t, d in sorted(run.items()):
        direct = [(o["function"], o["routes"]) for o in d["owners"] if o["routes"]]
        print("tip %s blob %s｜顶层函数 %d｜路由装饰器 %d｜Place 命中点 %d｜带路由的读站点函数 %d 枚"
              % (t[:8], d["main_py_blob"][:10], d["top_level_functions"], d["route_decorators_total"],
                 d["place_hit_points"], len(direct)))
        for name, routes in direct:
            print("    ", name, routes)
    dst = ART / ("place_hit_attribution_r1_%s.json" % tips[0][:8])
    dst.write_text(json.dumps({"booked_tip": booked, "runs": run}, ensure_ascii=False, indent=2)
                   + "\n", encoding="utf-8", newline=chr(10))
    print("ARTIFACT", dst)
    return 0


if __name__ == "__main__":
    sys.exit(main())
