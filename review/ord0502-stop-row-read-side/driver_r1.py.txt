"""读侧两问：哪条路由能读到站点行、行里哪个字段指回行程。
两半都要现读，不许从上一代外推：
  A 源级：`git grep -n` 找出 main.py 里所有读 Place 的行，逐行**向上**找它所属的顶层函数
     与该函数的装饰器路由（前两把尺都错过：一把把夹在路由之间的自由函数算进上一枚路由，
     一把用 ast 只认出 148 枚路由、把整枚 `GET /api/v1/places` 漏掉）。
  B 真跑：现读尖导出、私有新库起后端、一次 ensure 供出场景，然后把候选 GET 路由逐条打一遍，
     每条记状态、字节、答复里各关键字段出现次数，以及库侧三张表的列形状。
"""

import os

import time

from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
RUNNER = WT2 / "review/ord0502/run_ord0502_stack_r1.py"
ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
MAIN = "backend/app/main.py"
MODELS = "backend/app/db/models.py"
NEEDLES = ("select(Place)", "Place.id", "Place.city_code", "session.get(Place")
BAD: list[str] = []


def g(*a: str) -> str:
    r = subprocess.run(["git", *a], cwd=str(WT2), capture_output=True, text=True,
                       encoding="utf-8", errors="replace")
    if r.returncode != 0:
        BAD.append("GIT_FAIL %s rc=%d" % (" ".join(a)[:60], r.returncode))
    return (r.stdout or "")


def enclosing_route(lines, n):
    """第 n 行（1 基）向上找最近的顶层 def，再找它头上的路由装饰器。"""

    i = n - 1
    while i >= 0 and not re.match(r"^(async )?def ", lines[i]):
        i -= 1
    if i < 0:
        return None
    def_line = i + 1
    j = i - 1
    decs = []
    while j >= 0:
        m = re.match(r'^@app\.(get|post|patch|put|delete)\(\s*"([^"]+)"', lines[j])
        if m:
            decs.append((m.group(1).upper(), m.group(2), j + 1))
            j -= 1
            continue
        if lines[j].strip() == "" or lines[j].startswith(("@", "def ", "async ")):
            j -= 1
            continue
        break
    fn = re.sub(r"^(async )?def (\w+).*", r"\2", lines[i])
    return {"function": fn, "def_line": def_line,
            "routes": sorted(decs, key=lambda x: x[2], reverse=True)[:2]}


def key_counts(obj, keys, depth=0, bag=None):
    bag = {k: 0 for k in keys}
    stack = [obj]
    while stack:
        cur = stack.pop()
        if isinstance(cur, dict):
            for k, v in cur.items():
                if k in bag:
                    bag[k] += 1
                stack.append(v)
        elif isinstance(cur, list):
            stack.extend(cur)
    return bag


def shape(obj):
    if isinstance(obj, dict):
        return {"type": "object", "keys": sorted(obj.keys())[:14]}
    if isinstance(obj, list):
        first = obj[0] if obj else None
        return {"type": "array", "length": len(obj),
                "first_item": shape(first) if first is not None else None}
    return {"type": type(obj).__name__}


def main() -> int:
    tip = g("ls-remote", "--refs", "origin", MIRROR).split("\t")[0].strip()
    if len(tip) != 40:
        print("REFUSED: 读不到产品线尖")
        return 1
    src = g("show", "%s:%s" % (tip, MAIN))
    lines = src.split("\n")
    if len(lines) < 1000:
        print("REFUSED: main.py 读不通")
        return 1
    route_rows = {}
    for idx, ln in enumerate(lines, 1):
        for needle in NEEDLES:
            if needle in ln:
                e = enclosing_route(lines, idx)
                if not e:
                    BAD.append("找不到所属函数 @ %d" % idx)
                    continue
                key = (e["function"], e["def_line"])
                rec = route_rows.setdefault(key, {"function": e["function"],
                                                  "def_line": e["def_line"],
                                                  "routes": e["routes"], "needle_hits": []})
                rec["needle_hits"].append({"line": idx, "needle": needle,
                                           "text": ln.strip()[:110]})
    src_routes = sorted(route_rows.values(), key=lambda r: r["def_line"])

    spec = g("show", "%s:%s" % (tip, MODELS))
    place_cols = re.findall(r"^\s+(\w+)\s*(?::[^=]+)?=\s*Column", spec, re.M)
    blob_main = g("rev-parse", "%s:%s" % (tip, MAIN)).strip()
    blob_models = g("rev-parse", "%s:%s" % (tip, MODELS)).strip()

    spec_mod = importlib.util.spec_from_file_location("ord0502_runner", str(RUNNER))
    mod = importlib.util.module_from_spec(spec_mod)
    spec_mod.loader.exec_module(mod)
    run_id = "READSIDE-" + time.strftime("%H%M%S")
    snap = Path("D:/moodle") / ("_qoder_readside_%s" % run_id)
    snap.mkdir(parents=True, exist_ok=True)
    proc = subprocess.run(["git", "archive", "--format=tar", tip], cwd=str(WT2),
                          capture_output=True)
    if proc.returncode != 0:
        print("REFUSED: archive rc=%d" % proc.returncode)
        return 1
    tar = snap / "a.tar"
    tar.write_bytes(proc.stdout)
    import tarfile
    with tarfile.open(str(tar)) as t:
        t.extractall(path=str(snap))
    tar.unlink()
    disk_main = g("hash-object", "--no-filters", str(snap / MAIN)).strip()
    if disk_main != blob_main:
        BAD.append("导出的 main.py 字节不等于尖上那枚")
        print("REFUSED:", *BAD, sep="\n  ")
        return 1

    db = snap / "backend" / ("%s.db" % run_id)
    port = mod.free_port(int(os.environ.get("READSIDE_PORT", "4807")), "backend")
    env = mod.env_base(run_id, db, port)
    env["CITYNEXUS_DEMO_MODE"] = "true"
    token = secrets.token_hex(16)
    env["CITYNEXUS_DEMO_SCENE_TOKEN"] = token
    base = "http://127.0.0.1:%d" % port
    hdr = {"Content-Type": "application/json", "X-Demo-Scene-Token": token,
           "Idempotency-Key": "readside-" + run_id}
    out = {"run_id": run_id, "line_tip": tip, "blobs": {MAIN: blob_main, MODELS: blob_models},
           "backend_port": port, "db": db.as_posix(),
           "place_column_names_from_source": place_cols,
           "routes_reading_place_rows_from_source": src_routes}
    child = subprocess.Popen([mod.PY, "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1",
                              "--port", str(port)], cwd=str(snap / "backend"), env=env,
                             stdout=(ART / ("uvicorn_%s.log" % run_id)).open("w", encoding="utf-8"),
                             stderr=subprocess.STDOUT)
    out["child_pid"] = child.pid

    def get(path):
        try:
            with urllib.request.urlopen(base + path, timeout=60) as r:
                return r.status, r.read()
        except urllib.error.HTTPError as e:
            return e.code, e.read()
        except Exception as e:  # noqa: BLE001
            return None, (type(e).__name__ + str(e))[:120].encode()

    try:
        ok, reading = mod.wait_http(base + "/api/v1/health")
        hs, hb = get("/api/v1/health")
        out["health"] = {"ok": ok, "http_status": hs,
                         "carries_this_run_id": run_id in hb.decode("utf-8", "replace")}
        if not out["health"]["carries_this_run_id"]:
            BAD.append("归属未由本 run id 证")
        r = urllib.request.Request(base + "/api/v1/demo/scene/ensure", data=b"{}", method="POST",
                                   headers=hdr)
        try:
            with urllib.request.urlopen(r, timeout=90) as resp:
                rec = json.loads(resp.read().decode("utf-8"))
        except urllib.error.HTTPError as e:
            rec = {"http_error": e.code, "body": e.read()[:200].decode("utf-8", "replace")}
        jid = rec.get("journey_id")
        if not jid:
            BAD.append("ensure 没供出行程号")
            print("REFUSED:", *BAD, sep="\n  ")
            return 1
        pv = sqlite3.connect("file:%s?mode=ro" % db.as_posix(), uri=True).execute(
            "SELECT active_plan_version_id FROM journeys WHERE id=?", (jid,)).fetchone()[0]
        out["ensure"] = {"state": rec.get("state"), "journey_id": jid,
                         "journey_status": rec.get("journey_status")}
        pid = None
        with sqlite3.connect("file:%s?mode=ro" % db.as_posix(), uri=True) as con:
            row = con.execute("SELECT place_id FROM activities WHERE plan_version_id=? "
                              "AND place_id IS NOT NULL ORDER BY sequence_no LIMIT 1",
                              (str(pv),)).fetchone()
            pid = row[0] if row else None
            out["db_shapes"] = {}
            for t in ("places", "activities", "plan_versions", "journey_legs"):
                cols = [r[1] for r in con.execute('PRAGMA table_info("%s")' % t)]
                cnt = con.execute('SELECT count(*) FROM "%s"' % t).fetchone()[0]
                out["db_shapes"][t] = {"columns": cols, "rows": cnt,
                                       "has_journey_id": "journey_id" in cols,
                                       "has_plan_version_id": "plan_version_id" in cols,
                                       "has_place_id": "place_id" in cols}
                if t == "activities":
                    out["db_shapes"][t]["rows_on_active_plan_version"] = con.execute(
                        'SELECT count(*) FROM activities WHERE plan_version_id=?',
                        (str(pv),)).fetchone()[0]
        probes = ["/api/v1/places?city_code=CNX-DEMO",
                  "/api/v1/journeys/%s" % jid,
                  "/api/v1/journeys/%s/plans" % jid,
                  "/api/v1/journeys/%s/calendar.ics" % jid,
                  "/api/v1/places/%s" % pid if pid else "/api/v1/places/none",
                  "/api/v1/demo/scene"]
        res = []
        for p in probes:
            st, body = get(p)
            j = None
            try:
                j = json.loads(body.decode("utf-8"))
            except Exception:  # noqa: BLE001
                j = None
            rec2 = {"path": p.replace(jid, "<jid>"), "http_status": st, "bytes": len(body),
                    "content_is_json": j is not None}
            if j is not None:
                rec2["key_occurrences"] = key_counts(j, ("journey_id", "plan_version_id",
                                                         "place_id", "id", "title", "city_code"))
                sh = shape(j)
                rec2["shape"] = sh
                if sh.get("type") == "array" and sh.get("first_item"):
                    rec2["first_item_keys"] = sh["first_item"].get("keys")
            else:
                rec2["head"] = body[:90].decode("utf-8", "replace")
            res.append(rec2)
        out["read_side_probes"] = res
    finally:
        child.terminate()
        try:
            child.wait(timeout=60)
        except subprocess.TimeoutExpired:
            child.kill()
        out["child_exit"] = child.poll()

    places_probe = next((r for r in out["read_side_probes"]
                         if r["path"].startswith("/api/v1/places?")), None)
    if not places_probe or places_probe["http_status"] != 200:
        BAD.append("GET /api/v1/places 没答 200")
    elif not (places_probe.get("key_occurrences", {}).get("title", 0)
              and places_probe["key_occurrences"]["id"]):
        BAD.append("站点路由的答复里没有 id/title，'读到站点行'这句不成立")
    if out["db_shapes"]["places"]["has_journey_id"]:
        BAD.append("places 表这次带了 journey_id 列，块里那句要改")
    if not out["db_shapes"]["activities"]["has_plan_version_id"]:
        BAD.append("activities 没有 plan_version_id 列")
    if not any(r["function"] == "search_places" for r in src_routes):
        BAD.append("源级清单里没有 search_places，切法又错了")
    out["BAD"] = BAD
    dst = ART / ("place_read_side_%s.json" % run_id)
    dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + "\n", encoding="utf-8",
                   newline=chr(10))
    keep = {k: out[k] for k in ("line_tip", "run_id", "backend_port", "ensure", "db_shapes",
                                "read_side_probes")}
    print(json.dumps(keep, ensure_ascii=False, indent=2)[:3600])
    print("源级读到站点行的函数：",
          [(r["function"], r["def_line"], [x[1] for x in r["routes"]]) for r in src_routes])
    print("ARTIFACT", dst, "| BAD", BAD)
    return 1 if BAD else 0


if __name__ == "__main__":
    sys.exit(main())
