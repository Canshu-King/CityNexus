const fs = require("fs");
const { chromium } = require("playwright");
const BASE = process.env.ORD561_BASE || "http://127.0.0.1:4783";
const JID = process.env.ORD561_JID;
(async () => {
  const b = await chromium.launch();
  const ctx = await b.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 3,
    isMobile: true, hasTouch: true, locale: "en-US", serviceWorkers: "block" });
  const p = await ctx.newPage();
  await p.addInitScript(([jid, loc]) => { localStorage.setItem("citynexus.locale", loc);
    if (jid) localStorage.setItem("citynexus.v8.journey_id", jid); }, [JID || "", "en"]);
  await p.goto(BASE + "/v8/trips", { waitUntil: "domcontentloaded", timeout: 60000 });
  await p.waitForTimeout(2500);
  const r = await p.evaluate(() => {
    const d = document.querySelector(".cn-abort__form");
    const btn = document.querySelector('[data-testid="journey-abort-submit"]');
    const ta = document.querySelector(".cn-abort__input");
    const box = (el) => { if (!el) return null; const x = el.getBoundingClientRect();
      return { w: Math.round(x.width), h: Math.round(x.height), y: Math.round(x.y) }; };
    const cs = (el) => el ? getComputedStyle(el).display : null;
    return { details_open: d ? d.open : null, details_box: box(d), details_display: cs(d),
             summary_box: box(document.querySelector(".cn-abort__summary")),
             textarea_box: box(ta), textarea_display: cs(ta),
             button_box: box(btn), button_display: cs(btn),
             abort_section_box: box(document.querySelector(".cn-abort")),
             section_display: cs(document.querySelector(".cn-abort")) };
  });
  fs.writeFileSync(process.env.AS_OUT || "ord0561_abort_state_r1.json",
                 JSON.stringify(r, null, 2) + String.fromCharCode(10), "utf8");
  console.log(JSON.stringify(r, null, 1));
  await ctx.close(); await b.close();
})();
