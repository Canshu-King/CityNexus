# -*- coding: utf-8 -*-
"""ORD-0564 的全量扫描尺（只读：git show 现尖取字节，只打印，不写产品树）。

判据与号令者那把一致：值里含 CJK、含 ≥2 个拉丁词、且不以「。」结尾；每个 key 只取第一次出现
（= 英文侧；第二次是中文侧，会全命中，那是假阳性）。
比号令者那把多两格：
 ① 分母：扫的不只主字典，还有全树 `*Copy.ts`（组件自己的 copy 文件）；
 ② 每条命中都现读它的**读者数**（除定义文件外，还有谁引用这个键名），据此分「活键 / 死键」——
    死键修了屏上也不会变，活键才是用户能感觉到的那一格。
"""
import json
import re
import subprocess
import sys
from collections import OrderedDict

sys.stdout.reconfigure(encoding="utf-8", errors="replace")
WT = "D:/moodle/_qoder_newlane_ord0444_20260929/wt2"
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
CJK = re.compile(r"[\u3400-\u4dbf\u4e00-\u9fff\u3040-\u30ff\u3000-\u303f\uff00-\uffef]")
LATIN_WORDS = re.compile(r"[A-Za-z]{2,}")
KEY = re.compile(r'^\s*([A-Za-z][A-Za-z0-9_.]*)\s*:\s*(["\'])(.*?)\2\s*,?\s*(?://.*)?$', re.S)


def sh(*a):
    return subprocess.run(list(a), cwd=WT, capture_output=True, text=True,
                          encoding="utf-8", errors="replace")


def tip():
    out = sh("git", "ls-remote", "--refs", "origin", MIRROR).stdout
    return out.split("\t")[0].strip()


def copy_files(t):
    out = sh("git", "ls-tree", "-r", "--name-only", t, "--", "frontend/src").stdout
    files = [f for f in out.splitlines() if f.endswith("Copy.ts") or
             f.endswith("LocaleContext.tsx")]
    return sorted(files)


def scan_file(t, path):
    body = sh("git", "show", "%s:%s" % (t, path)).stdout
    hits, seen = [], OrderedDict()
    for i, line in enumerate(body.split("\n"), 1):
        m = KEY.match(line)
        if not m:
            continue
        key, val = m.group(1), m.group(3)
        if key in seen:
            seen[key] += 1
            continue                      # 第二次出现 = 中文侧，按判据跳过
        seen[key] = 1
        if not CJK.search(val):
            continue
        if len(LATIN_WORDS.findall(val)) < 2:
            continue
        if val.rstrip().endswith("。"):
            continue
        hits.append({"file": path, "line": i, "key": key, "en_value": val,
                     "cjk_chars": "".join(sorted(set(CJK.findall(val)))),
                     "latin_words": len(LATIN_WORDS.findall(val))})
    return hits, len(seen)


def readers(t, key, own_file):
    """除定义文件外，还有谁在引用这个键名（含 t("key") / 字典名.key / 测试）。"""
    out = sh("git", "grep", "-n", "--fixed-strings", key, t, "--", "frontend", "e2e",
             "tests").stdout
    rows = []
    for line in out.splitlines():
        rest = line.split(":", 2)
        if len(rest) < 3:
            continue
        path = rest[1]
        if path == own_file:
            continue
        rows.append({"path": path, "line": rest[2].split(":")[0], "text": rest[2].split(":", 1)[-1][:110]})
    return rows


def main():
    t = tip()
    files = copy_files(t)
    report = {"line_tip": t, "files_scanned": len(files), "files": files, "hits": [],
              "keys_scanned": 0}
    for f in files:
        hits, n = scan_file(t, f)
        report["keys_scanned"] += n
        for h in hits:
            rs = readers(t, h["key"], f)
            h["readers_outside_definition"] = len(rs)
            h["reader_sample"] = rs[:4]
            h["class"] = "LIVE" if rs else "DEAD_KEY_ZERO_READER"
            report["hits"].append(h)
    print(json.dumps(report, ensure_ascii=False, indent=2))
    live = [h for h in report["hits"] if h["class"] == "LIVE"]
    dead = [h for h in report["hits"] if h["class"] != "LIVE"]
    sys.stderr.write("tip %s | files %d | keys %d | hits %d (LIVE %d / DEAD %d)\n"
                     % (t[:12], len(files), report["keys_scanned"], len(report["hits"]),
                        len(live), len(dead)))
    for h in report["hits"]:
        sys.stderr.write("  %-6s %-52s :%s  %s\n" % (h["class"], h["file"] + ":" + str(h["line"]),
                                                      h["key"], h["en_value"][:96]))
    return 0


if __name__ == "__main__":
    sys.exit(main())
