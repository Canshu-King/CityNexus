"""ORD-0564 反件走查 r2：在**自己那台私有栈**上把 City Discovery 屏两臂各读一遍。

代次钉法与停靠行那一手相同：入口资产「被服务字节 == 磁盘字节 == 构建记录字节」逐字比，
构建记录就是改完字节后重跑 vite 那趟（ord0564_build_r1.json）。"""
from __future__ import annotations

import json
import os
import socket
import subprocess
import sys
import threading
import time
import urllib.error
import urllib.request
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
SCR = Path("D:/moodle/_qoder_newlane_ord0444_20260929/scratch")
WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
BUILD = json.loads((ART / "stoprow_build_state_r1.json").read_text(encoding="utf-8"))
PATCHED = json.loads((ART / "ord0564_build_r1.json").read_text(encoding="utf-8"))
SNAP = Path(BUILD["snap"])
DIST = SNAP / "frontend" / "dist"
PROBE = SCR / "ord0564_probe_r1.cjs"
SHOTS = Path("D:/moodle/_qoder_ord0561_shots")
PYX = Path("D:/moodle/CityNexus/.venv/Scripts/python.exe")
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
FORBIDDEN = {8000, 4781, 4782, 4783, 4785, 4787, 4789, 8693, 4799}
HAN = "一-鿿"
BAD: list[str] = []


def sha1(b: bytes) -> str:
    return hashlib.sha1(b).hexdigest()


def free_port(wanted: int, taken=()) -> int:
    p = wanted
    while p < wanted + 60:
        if p not in FORBIDDEN and p not in taken:
            with socket.socket() as s:
                try:
                    s.bind(("127.0.0.1", p))
                    return p
                except OSError:
                    pass
        p += 1
    raise SystemExit("找不到空闲私有端口")


def http(url, data=None, headers=None, timeout=60):
    req = urllib.request.Request(url, data=data, method="POST" if data is not None else "GET",
                                 headers=headers or {})
    try:
        with urllib.request.urlopen(req, timeout=timeout) as r:
            return r.status, r.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()
    except Exception as e:  # noqa: BLE001
        return None, (type(e).__name__ + str(e))[:150].encode()


def main() -> int:
    tip = subprocess.run(["git", "ls-remote", "--refs", "origin", MIRROR], cwd=str(WT2),
                         capture_output=True, text=True).stdout.split("\t")[0].strip()
    run_id = "ORD564-%s" % time.strftime("%H%M%S")
    db = SNAP / "backend" / ("%s.db" % run_id)
    api = free_port(4841)
    front = free_port(api + 5, taken={api})
    env = dict(os.environ)
    for k in ("HTTP_PROXY", "HTTPS_PROXY", "http_proxy", "https_proxy"):
        env.pop(k, None)
    env.update({"CITYNEXUS_DATABASE_URL": "sqlite:///%s" % db.as_posix(),
                "CITYNEXUS_E2E_RUN_ID": run_id, "CITYNEXUS_BACKEND_PORT": str(api),
                "CITYNEXUS_DEMO_MODE": "true",
                "CITYNEXUS_DEMO_SCENE_TOKEN": os.urandom(16).hex()})
    token = env["CITYNEXUS_DEMO_SCENE_TOKEN"]
    out = {"run_id": run_id, "driver": Path(__file__).name, "probe": PROBE.name,
           "snap": str(SNAP), "snap_build_tip": BUILD["line_tip"], "line_tip_at_walk": tip,
           "api_port": api, "front_port": front,
           "token_names_only": ["CITYNEXUS_DEMO_SCENE_TOKEN", "X-Demo-Scene-Token"]}
    log_path = ART / ("uvicorn_%s.log" % run_id)
    backend = subprocess.Popen([str(PYX), "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1",
                                "--port", str(api)], cwd=str(SNAP / "backend"), env=env,
                               stdout=log_path.open("w", encoding="utf-8"),
                               stderr=subprocess.STDOUT)
    out["backend_pid"] = backend.pid
    served = {}

    class H(BaseHTTPRequestHandler):
        def log_message(self, *a):
            pass

        def _send(self, code, body, ctype):
            self.send_response(code)
            self.send_header("Content-Type", ctype)
            self.send_header("Content-Length", str(len(body)))
            self.send_header("Cache-Control", "no-store")
            self.end_headers()
            try:
                self.wfile.write(body)
            except (ConnectionAbortedError, ConnectionResetError, BrokenPipeError):
                pass

        def do_GET(self):
            p = self.path.split("?")[0]
            if p.startswith("/api/"):
                st, body = http("http://127.0.0.1:%d%s" % (api, self.path))
                self._send(st or 502, body, "application/json")
                return
            f = (DIST / (p.lstrip("/") or "index.html")).resolve()
            ctype = ("text/javascript" if f.suffix == ".js" else "text/css" if f.suffix == ".css"
                     else "text/html" if f.suffix == ".html" else "application/octet-stream")
            if not str(f).startswith(str(DIST)) or not f.is_file():
                f, ctype = DIST / "index.html", "text/html"
            self._send(200, f.read_bytes(), ctype)

        do_POST = do_GET

    srv = ThreadingHTTPServer(("127.0.0.1", front), H)
    threading.Thread(target=srv.serve_forever, daemon=True).start()
    try:
        ok = False
        for _ in range(90):
            st, body = http("http://127.0.0.1:%d/api/v1/health" % api)
            if st == 200 and run_id.encode() in body:
                ok = True
                break
            time.sleep(1)
        out["health_via_api"] = {"ok": ok, "carries_run_id": ok}
        if not ok:
            BAD.append("后端没答或答复里没本 run id（日志 %s）" % log_path.name)
            raise SystemExit(1)
        st, body = http("http://127.0.0.1:%d/api/v1/health" % front)
        if run_id.encode() not in body:
            BAD.append("经前端原点没证到归属（代理指向别处）")
        hdr = {"Content-Type": "application/json", "X-Demo-Scene-Token": token,
               "Idempotency-Key": "ord564-" + run_id}
        req = urllib.request.Request("http://127.0.0.1:%d/api/v1/demo/scene/ensure" % api,
                                     data=b"{}", headers=hdr, method="POST")
        with urllib.request.urlopen(req, timeout=120) as r:
            rec = json.loads(r.read().decode("utf-8"))
        jid = rec.get("journey_id")
        out["ensure"] = {"http_status": r.status, "state": rec.get("state")}
        if not jid or rec.get("state") != "READY":
            BAD.append("供不出场景，屏读没有前置")
            raise SystemExit(1)
        # 那一屏要的是"这枚行程名下还没有架子"，所以先问一次集合面，把答复也留下
        st, body = http("http://127.0.0.1:%d/api/v1/collections" % front)
        out["collections_http_status"] = st
        out["collections_is_list"] = isinstance(json.loads(body), list) if st == 200 else None

        for a in PATCHED["assets"]:
            if a["name"].startswith("index-") and a["name"].endswith((".js", ".css")):
                st, body = http("http://127.0.0.1:%d/assets/%s" % (front, a["name"]))
                disk = (DIST / "assets" / a["name"]).read_bytes()
                served[a["name"]] = {"http_status": st, "bytes": len(body),
                                     "served_sha1": sha1(body), "disk_sha1": sha1(disk),
                                     "served_equals_disk": body == disk,
                                     "disk_equals_patched_record": sha1(disk) == a["sha1"]}
        out["byte_pairing"] = served
        if not served or not all(v["served_equals_disk"] and v["disk_equals_patched_record"]
                                 for v in served.values()):
            BAD.append("被服务字节与磁盘/改后构建记录不逐字等 —— 屏读不能归到我改的那一代")
        entry_js = [n for n in served if n.endswith(".js")]
        out["entry_asset_changed_vs_pre_fix"] = {
            "pre_fix": [a["name"] for a in BUILD["assets"] if a["name"].endswith(".js")],
            "post_fix": entry_js}

        dest = ART / ("ord0564_screen_%s.json" % run_id)
        penv = dict(os.environ)
        penv.update({"NODE_PATH": str(WT2 / "frontend" / "node_modules"),
                     "ORD564_FRONT": "http://127.0.0.1:%d" % front,
                     "ORD564_OUT": str(dest), "ORD564_SHOTS": str(SHOTS), "ORD564_JID": jid})
        node = subprocess.run(["node", str(PROBE)], capture_output=True, text=True,
                              encoding="utf-8", errors="replace", env=penv, timeout=420)
        out["probe_rc"] = node.returncode
        out["probe_stdout"] = (node.stdout or "")[-500:]
        out["probe_stderr"] = (node.stderr or "")[-400:]
        rep = json.loads(dest.read_text(encoding="utf-8")) if dest.exists() else {}
        arms = {a["locale"]: a for a in rep.get("arms") or []}
        out["screen"] = {"arms": rep.get("arms"), "errors": rep.get("errors")}
        if rep.get("errors"):
            BAD.append("探针自报 errors：%s" % rep["errors"])
        for need in ("en", "zh-CN"):
            if need not in arms:
                BAD.append("臂没跑到：%s" % need)
        if not BAD:
            en, zh = arms["en"], arms["zh-CN"]
            for a, want in ((en, "/v8/discover"), (zh, "/v8/discover")):
                if not str(a.get("url", "")).endswith(want):
                    BAD.append("这一臂没停在那一屏：%s" % a.get("url"))
            sen = " ".join(en.get("empty_state_lines") or [])
            szh = " ".join(zh.get("empty_state_lines") or [])
            out["verdict"] = {
                "en_sentence_has_city_discovery": "City Discovery" in sen,
                "en_sentence_han_count": len([c for c in sen if "一" <= c <= "鿿"]),
                "en_screen_han_line_count": en.get("han_chars_in_whole_screen"),
                "en_screen_han_lines": en.get("han_lines"),
                "zh_sentence_keeps_the_chinese_copy": "城市探索" in szh,
                "en_html_lang": en.get("html_lang"), "zh_html_lang": zh.get("html_lang"),
                "en_stored_locale": en.get("stored_locale"), "zh_stored_locale": zh.get("stored_locale"),
            }
            v = out["verdict"]
            if not v["en_sentence_has_city_discovery"]:
                BAD.append("en 臂那句没读到 City Discovery：%r" % sen[:90])
            if v["en_sentence_han_count"] != 0:
                BAD.append("en 臂那句里还有汉字")
            if v["en_screen_han_line_count"] != 0:
                BAD.append("en 臂整屏还有带汉字的行：%s" % v["en_screen_han_lines"][:3])
            if not v["zh_sentence_keeps_the_chinese_copy"]:
                BAD.append("zh 臂那句被我改坏了")
    except SystemExit:
        pass
    finally:
        srv.shutdown()
        srv.server_close()
        backend.terminate()
        try:
            backend.wait(timeout=40)
        except subprocess.TimeoutExpired:
            backend.kill()
        out["backend_exit"] = backend.poll()
    out["BAD"] = BAD
    dst = ART / ("ord0564_walk_%s.json" % run_id)
    dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + chr(10), encoding="utf-8",
                   newline=chr(10))
    print(json.dumps({k: out[k] for k in ("run_id", "probe", "snap_build_tip", "line_tip_at_walk",
                                          "api_port", "front_port", "ensure", "collections_http_status",
                                          "entry_asset_changed_vs_pre_fix", "verdict", "BAD")
                      if k in out}, ensure_ascii=False, indent=2))
    print("字节对照", json.dumps({k: [v["bytes"], v["served_equals_disk"],
                                      v["disk_equals_patched_record"]]
                                  for k, v in served.items()}, ensure_ascii=False))
    print("probe", out.get("probe_stdout", "").strip())
    print("ARTIFACT", dst)
    return 1 if BAD else 0


if __name__ == "__main__":
    sys.exit(main())
