/* 收藏状态唯一性那栋楼 · 第二格（真跑腿）：真点一次收藏，读接口与屏上两处表达，
   再真试一次"取消收藏"，把每步两处表达逐字记下来。

   图纸（账本 16603）：收藏从点下去到再看到它，状态必须唯一——不许"已收藏但列表里没有"
   或"列表里有但接口说没有"。判据自己能量：真跑一遍收藏与取消收藏，每步读接口与屏上两处表达。

   这一腿要落的读数：
   - 收藏前：接口列几行、屏上几行。
   - 点收藏后：接口 item_count 加没加、屏上 held 标有没有出现、屏上那句原话。
   - 找"取消"：屏上有没有一扇可点的取消门（枚举所有 collect/collection 相关控件）。
   - 试取消：能取消吗（屏上有没有门 / 接口有没有门）。
   - 判一致性：屏上"已在这"与接口"在册"是否逐字一致。

   一次写请求都不多发：只点收藏那一次；"取消"那一腿只读屏与接口，不构造写。 */
const { chromium } = require("playwright");

const BASE = process.env.RK_BASE;
const JID = process.env.RK_JID;
const OUT = process.env.RK_OUT;
const SHOTS = process.env.RK_SHOTS;
const WIDTH = parseInt(process.env.RK_WIDTH || "390", 10);
const ARM = process.env.RK_ARM || "zh-CN";

(async () => {
  const doc = { run_id: process.env.RK_RUN || "", arm: ARM, width: WIDTH, steps: [], errors: [] };
  const browser = await chromium.launch({ timeout: 120000 });
  try {
    const ctx = await browser.newContext({
      viewport: { width: WIDTH, height: 844 }, deviceScaleFactor: 2,
      isMobile: true, hasTouch: true, serviceWorkers: "block", locale: ARM,
    });
    const page = await ctx.newPage();
    const perr = [];
    page.on("pageerror", (e) => perr.push(String(e).slice(0, 160)));
    await page.goto(BASE + "/v8/hub", { waitUntil: "networkidle", timeout: 60000 });
    await page.evaluate(([a, j]) => {
      localStorage.setItem("citynexus.locale", a);
      localStorage.setItem("citynexus.v8.journey_id", j);
    }, [ARM, JID]);
    // 收藏这扇门挂在一枚地点上（POI 导入或行程里的地点）。先打开行程那一屏找 collect-entry。
    await page.goto(BASE + "/v8/trips", { waitUntil: "networkidle", timeout: 60000 });
    // 找第一枚 collect-entry（点开才有的层）
    let entry = page.locator("[data-testid='collect-entry']").first();
    if (!(await entry.count())) {
      // 试 poiimport 那一屏
      await page.goto(BASE + "/v8/bookings", { waitUntil: "networkidle", timeout: 60000 }).catch(() => {});
      entry = page.locator("[data-testid='collect-entry']").first();
    }
    if (!(await entry.count())) {
      doc.errors.push("屏上找不到 collect-entry（试了 trips/bookings）");
    } else {
      const read = async (label) => {
        const screen = await page.evaluate(() => {
          const held = Array.from(document.querySelectorAll("[data-testid='collect-held-mark']"))
            .map((n) => n.textContent.trim());
          const shelves = Array.from(document.querySelectorAll("[data-testid='collect-shelf']"))
            .map((n) => ({ name: n.textContent.trim(), already: n.getAttribute("data-already-held") }));
          const result = document.querySelector("[data-testid='collect-result']");
          const entryBtn = document.querySelector("[data-testid='collect-entry']");
          return {
            held_marks: held,
            shelves,
            result_text: result ? result.textContent.trim() : null,
            entry_aria: entryBtn ? entryBtn.getAttribute("aria-label") : null,
            // 枚举所有收藏相关的可点控件，找"取消/移除"门
            clickable: Array.from(document.querySelectorAll("button,a[role='button']"))
              .filter((n) => /collect|collection/i.test(n.getAttribute("data-testid") || ""))
              .map((n) => n.getAttribute("data-testid")),
            any_uncollect_label: Array.from(document.querySelectorAll("button,a[role='button']"))
              .map((n) => (n.getAttribute("aria-label") || n.textContent || "").trim())
              .filter((t) => /取消收藏|移出|删除收藏|移除收藏|uncollect|remove/i.test(t)),
          };
        });
        return { label, screen };
      };
      doc.steps.push(await read("before"));
      // 点收藏入口 → 打开挑层
      await entry.click({ timeout: 15000 });
      await page.waitForTimeout(600);
      doc.steps.push(await read("opened_picker"));
      // traveler_id 从行程上读（屏上就是这么拿的），再建一枚挑层——否则挑层是空的、无从点收藏。
      const tid = await page.evaluate(async () => {
        const jid = localStorage.getItem("citynexus.v8.journey_id");
        const r = await fetch("/api/v1/journeys/" + jid);
        const j = await r.json().catch(() => null);
        return (j && (j.traveler_id || (j.journey && j.journey.traveler_id))) || null;
      });
      doc.traveler_id = tid;
      if (!tid) { doc.errors.push("读不到 traveler_id"); }
      const created = await page.evaluate(async (t) => {
        const r = await fetch("/api/v1/collections?traveler_id=" + encodeURIComponent(t), {
          method: "POST", headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ name: "WalkShelf", owner_traveler_id: t }),
        });
        const j = await r.json().catch(() => null);
        return { status: r.status, id: j && j.id, name: j && j.name, body: j };
      }, tid);
      doc.created_shelf = created;
      // 再读接口（列表，带 traveler_id）——这才是屏上那扇门读的那一次
      const api = await page.evaluate(async (t) => {
        const r = await fetch("/api/v1/collections?traveler_id=" + encodeURIComponent(t), { headers: { Accept: "application/json" } });
        const j = await r.json().catch(() => null);
        return { status: r.status, shelves: Array.isArray(j) ? j.map((f) => ({ name: f.name, item_count: (f.items || []).length })) : j };
      }, tid);
      doc.steps.push({ label: "api_list", api });
      // 建层之后挑层要重新读一次；但 collect-entry 是开关，再点一下会把挑层收起来。
      // 所以：只在挑层当前是关着的时候才点它。
      const pickerOpen = await page.locator("[data-testid='collect-picker']").count();
      if (!pickerOpen) {
        await page.locator("[data-testid='collect-entry']").first().click({ timeout: 15000 }).catch(() => {});
        await page.waitForTimeout(700);
      } else {
        // 挑层已经开着：让它重读列表（屏上那扇门自己会读）
        await page.locator("[data-testid='collect-entry']").first().evaluate((n) => n.click()).catch(() => {});
        await page.waitForTimeout(700);
        await page.locator("[data-testid='collect-entry']").first().evaluate((n) => n.click()).catch(() => {});
        await page.waitForTimeout(700);
      }
      doc.steps.push(await read("picker_with_shelf"));
      // 点第一枚挑层（= 真发一次收藏）
      const shelf = page.locator("[data-testid='collect-shelf']").first();
      if (await shelf.count()) {
        await shelf.click({ timeout: 15000 });
        await page.waitForTimeout(800);
        doc.steps.push(await read("after_collect_click"));
      } else {
        doc.errors.push("挑层里没有 collect-shelf");
      }
      await page.screenshot({ path: SHOTS + "/collect_after_" + WIDTH + "_" + ARM + ".png" }).catch(() => {});
      doc.page_errors = perr.slice(0, 5);
    }
    await ctx.close();
  } catch (e) {
    doc.errors.push({ fatal: String(e).slice(0, 300) });
  } finally { await browser.close(); }
  fs = require("fs");
  fs.writeFileSync(OUT, JSON.stringify(doc, null, 1) + "\n", "utf-8");
  console.log("steps=" + doc.steps.length + " errors=" + doc.errors.length +
    " page_errors=" + ((doc.page_errors || []).length));
  for (const s of doc.steps) {
    if (s.screen) console.log("  [" + s.label + "] held=" + s.screen.held_marks.length +
      " shelves=" + s.screen.shelves.length + " uncollect_labels=" + s.screen.any_uncollect_label.length +
      " result=" + String(s.screen.result_text || "").slice(0, 40));
    if (s.api) console.log("  [api_list] status=" + s.api.status + " shelves=" + JSON.stringify(s.api.shelves));
  }
  for (const e of doc.errors) console.log("  ERR " + JSON.stringify(e).slice(0, 120));
  process.exit(0);
})();
