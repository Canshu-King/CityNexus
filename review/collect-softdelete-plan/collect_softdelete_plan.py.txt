def class_cols(src, name):
    seg = src[src.index("class %s(" % name):]
    end = seg.find("class ", 10)
    seg = seg[:end] if end > 0 else seg
    cols = {}
    for l in seg.splitlines():
        m = re.match(r"\s*(\w+): Mapped", l)
        if m:
            cols[m.group(1)] = True
    return cols
