// 1ba48dbf · 模拟悬浮件的压覆盖几何量：只量矩形与命中，全程不点击、不触发任何写路径
// 要回答的问题很窄：那一枚 fixed 悬浮件（div#sim-float / button#sim-ball）在哪些屏、哪些档、
// 哪些滚动位置上，会把已渲染的交互控件的中心点盖在自己下面。
const fs = require("fs");
const { chromium } = require("playwright");

const BASE = process.env.CL_BASE;
const OUT = process.env.CL_OUT;
const SHOTDIR = process.env.CL_SHOTS;
const JID = process.env.CL_JID;
const WIDTHS = (process.env.CL_WIDTHS || "360,390,414").split(",").map(Number);
const ROUTES = (process.env.CL_ROUTES || "hub,trips").split(",");

const GEOM = () => {
  const de = document.scrollingElement || document.documentElement;
  const vh = window.innerHeight, vw = window.innerWidth;
  const floats = [];
  for (const el of Array.from(document.querySelectorAll("#sim-float, #sim-ball, .cn-sim-float, .cn-sim-ball"))) {
    const r = el.getBoundingClientRect();
    if (r.width <= 0 || r.height <= 0) { continue; }
    const cs = getComputedStyle(el);
    floats.push({ el: el, id: el.id || "", cls: (typeof el.className === "string"
      ? el.className.trim().split(/\s+/)[0] : ""),
      tag: el.tagName.toLowerCase(),
      rect: { top: Math.round(r.top), bottom: Math.round(r.bottom),
              left: Math.round(r.left), right: Math.round(r.right) },
      pos: cs.position, z: cs.zIndex, pe: cs.pointerEvents,
      aria: el.getAttribute("aria-label") || "", title: el.getAttribute("title") || "",
      text: (el.textContent || "").replace(/\s+/g, " ").trim().slice(0, 30),
      kids: el.children.length,
      closers: Array.from(el.querySelectorAll("button,[role=button]"))
        .map((b) => (b.getAttribute("data-testid") || "") + "「"
          + (b.textContent || "").replace(/\s+/g, " ").trim().slice(0, 16) + "」"
          + (b.getAttribute("aria-label") ? "[aria:" + b.getAttribute("aria-label") + "]" : ""))
    });
  }
  const inside = (pt, rc) => pt.x >= rc.left && pt.x <= rc.right && pt.y >= rc.top && pt.y <= rc.bottom;
  const hit = [];
  for (const el of Array.from(document.querySelectorAll(
      "button,a,summary,[role=button],[role=link],input,select,textarea"))) {
    const r = el.getBoundingClientRect();
    if (r.width <= 0 || r.height <= 0) { continue; }
    const cx = Math.round(r.left + r.width / 2), cy = Math.round(r.top + r.height / 2);
    if (cx < 0 || cx > vw || cy < 0 || cy > vh) { continue; }
    const top = document.elementFromPoint(cx, cy);
    if (top && (top === el || el.contains(top) || top.contains(el))) { continue; }
    const owner = floats.find((f) => inside({ x: cx, y: cy }, f.rect));
    hit.push({ tag: el.tagName.toLowerCase(), testid: el.getAttribute("data-testid") || "",
               aria: el.getAttribute("aria-label") || "",
               text: (el.textContent || "").replace(/\s+/g, " ").trim().slice(0, 26),
               rect: Math.round(r.top) + ".." + Math.round(r.bottom)
                 + " × " + Math.round(r.left) + ".." + Math.round(r.right),
               center: cx + "," + cy,
               by: top ? (top.tagName.toLowerCase() + (top.id ? "#" + top.id : "")
                 + (typeof top.className === "string" && top.className
                   ? "." + top.className.trim().split(/\s+/)[0] : "")) : null,
               on_float: !!owner,
               float_id: owner ? (owner.id || owner.cls) : "" });
    }
  return { vh: vh, vw: vw, doc_h: de.scrollHeight,
           max_scroll: de.scrollHeight - de.clientHeight, scroll_y: Math.round(window.scrollY),
           floats: floats.map((f) => ({ id: f.id, cls: f.cls, tag: f.tag, rect: f.rect,
                                        pos: f.pos, z: f.z, pe: f.pe, aria: f.aria,
                                        title: f.title, text: f.text, kids: f.kids,
                                        closers: f.closers })),
           covered: hit, rendered: Array.from(document.querySelectorAll(
             "button,a,summary,[role=button],[role=link],input,select,textarea"))
             .filter((e) => { const r = e.getBoundingClientRect(); return r.width > 0 && r.height > 0; }).length };
};

(async () => {
  const browser = await chromium.launch();
  const rep = { base: BASE, widths: WIDTHS, routes: ROUTES, cells: [], errors: [] };
  const save = () => fs.writeFileSync(OUT, JSON.stringify(rep, null, 1) + "\n", "utf8");
  for (const arm of ["zh-CN", "en"]) {
    for (const route of ROUTES) {
      for (const vw of WIDTHS) {
        const ctx = await browser.newContext({ viewport: { width: vw, height: 844 },
          deviceScaleFactor: 2, isMobile: true, hasTouch: true,
          locale: arm === "en" ? "en-US" : "zh-CN", serviceWorkers: "block" });
        const page = await ctx.newPage();
        const rec = { arm: arm, route: route, vw: vw, states: [] };
        try {
          await page.addInitScript(([jid, loc]) => {
            try {
              if (jid) { localStorage.setItem("citynexus.v8.journey_id", jid); }
              localStorage.setItem("citynexus.locale", loc);
            } catch (e) {}
          }, [JID || "", arm]);
          await page.goto(BASE + "/v8/" + route, { waitUntil: "domcontentloaded", timeout: 60000 });
          await page.waitForTimeout(3400);
          rec.href = page.url();
          const OPEN = (process.env.CL_OPEN || "").split(",").filter(Boolean);
          if (OPEN.length) {
            for (const tid of OPEN) {
              const t = page.locator("[data-testid='" + tid + "']").first();
              rec.opened = rec.opened || [];
              const isDet = await t.evaluate((el) => el.tagName === "DETAILS").catch(() => false);
              const clicker = isDet ? page.locator("[data-testid='" + tid + "'] > summary").first() : t;
              await clicker.scrollIntoViewIfNeeded().catch(() => {});
              await clicker.click({ timeout: 8000 })
                .catch((e) => { rec.open_error = (e.message || "").slice(0, 70); });
              rec.opened.push(tid);
            }
            await page.waitForTimeout(800);
          }
          for (const pos of ["top", "max"]) {
            if (pos === "max") {
              await page.evaluate(() => window.scrollTo(0, 99999));
              await page.waitForTimeout(350);
            }
            const g = await page.evaluate(GEOM);
            g.state = pos + (OPEN.length ? "_opened" : "");
            rec.states.push(g);
            const fl = g.floats[0];
            console.log(rec.arm + " " + rec.route + " w" + rec.vw + " " + pos
              + " | 悬浮件在场=" + g.floats.length
              + (fl ? " rect " + fl.rect.top + ".." + fl.rect.bottom + " × " + fl.rect.left + ".."
                + fl.rect.right + " pos=" + fl.pos + " z=" + fl.z + " aria=\"" + fl.aria
                + "\" 关闭件=" + JSON.stringify(fl.closers) : "")
              + " | 屏上控件 " + g.rendered + " 枚、中心被盖 " + g.covered.length + " 枚"
              + "（其中落在悬浮件矩形里 " + g.covered.filter((x) => x.on_float).length + " 枚）"
              + " | 滚位 " + g.scroll_y + "/" + g.max_scroll);
            for (const x of g.covered.filter((y) => y.on_float).slice(0, 5)) {
              console.log("    盖 " + x.tag + "#" + x.testid + "「" + x.text + "」 中心 "
                + x.center + " 顶层 " + x.by + " 悬浮件 " + x.float_id);
            }
          }
          await page.screenshot({ path: SHOTDIR + "/f_" + rec.route + "_" + vw + "_" + arm + ".png" })
            .catch(() => {});
        } catch (e) {
          rep.errors.push(arm + "/" + route + "/" + vw + " " + (e.message || String(e)).slice(0, 110));
        }
        rep.cells.push(rec);
        save();
        await ctx.close().catch(() => {});
      }
    }
  }
  await browser.close();
  save();
  const all = [];
  for (const c of rep.cells) { for (const s of c.states) { all.push({ c: c, s: s }); } }
  console.log("cells=" + rep.cells.length + " 态=" + all.length
    + " 悬浮件在场态=" + all.filter((a) => a.s.floats.length > 0).length
    + " 压住控件的态=" + all.filter((a) => a.s.covered.some((x) => x.on_float)).length
    + " 被压控件读数=" + all.reduce((n, a) => n + a.s.covered.filter((x) => x.on_float).length, 0)
    + " errors=" + rep.errors.length);
})();
