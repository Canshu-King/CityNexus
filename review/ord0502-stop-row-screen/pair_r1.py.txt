# -*- coding: utf-8 -*-
"""把两趟 r6 走查的读数并排打出来，并逐格判"两趟是否同值"。块里凡引这两趟的数，都从这里取。"""
import io
import json
import sys

sys.stdout.reconfigure(encoding="utf-8", errors="replace")
paths = sys.argv[1:]
ds = [json.load(io.open(p, encoding="utf-8")) for p in paths]


def key(d):
    sc = d["screen"]
    return {
        "run_id": d["run_id"], "probe": d["probe"], "driver": d.get("driver"),
        "tip_at_walk": d["line_tip_at_walk"], "build_tip": d["build_tip"],
        "api_port": d["api_port"], "front_port": d["front_port"],
        "ensure": d["ensure"], "api_stop_count": d["api_stop_count"],
        "titles": d["api_stop_titles"], "place_ids": d["api_stop_place_ids"],
        "fixture": d["fixture"],
        "byte_pairing": {k: [v["bytes"], v["served_sha1"], v["disk_sha1"],
                             v["served_equals_disk"], v["disk_equals_build_record"]]
                         for k, v in d["byte_pairing"].items()},
        "frame": sc["frame"],
        "bar": [sc["bar_band"]["count"], sc["bar_band"]["top"],
                [(i["tag"], i["cls"], i["rect"], i["z"]) for i in sc["bar_band"]["items"]]],
        "static": [[r["index"], r["rect"]["y"], r["rect"]["bottom"], r["rect"]["height"],
                    r["rect"]["width"], r["center_in_viewport"], r["text"]] for r in sc["static_rows"]],
        "if_needed": [[r["index"], r["scroll_y"], r["rect"]["y"], r["rect"]["bottom"],
                       r["under_bar"], (r["top_of_stack"] or {}).get("cls"),
                       (r["top_of_stack"] or {}).get("testid")] for r in sc["if_needed_rows"]],
        "above_bar": [[r["index"], r["scroll_y"], r["rect"]["y"], r["rect"]["bottom"],
                       r["clears_bar"], (r["top_of_stack"] or {}).get("cls"),
                       (r["top_of_stack"] or {}).get("testid")] for r in sc["above_bar_rows"]],
        "counts": sc["counts"], "errors": sc["errors"], "BAD": d["BAD"],
        "screenshot": sc["screenshot"],
        "triple": d["name_triple_check"],
    }


ks = [key(d) for d in ds]
for k in ks:
    print(json.dumps(k, ensure_ascii=False, sort_keys=True))
same = all(json.dumps(x, sort_keys=True) == json.dumps(ks[0], sort_keys=True)
           for x in ks[1:])
print("SAME_APART_FROM_RUN_AND_PORTS:", same)
diff = [f for f in ("ensure", "api_stop_count", "frame", "bar", "static", "if_needed", "above_bar",
                    "counts", "errors", "BAD", "triple", "byte_pairing", "fixture", "titles",
                    "place_ids")
        if len({json.dumps(x[f], sort_keys=True) for x in ks}) != 1]
print("FIELDS_DIFFERING:", diff)
