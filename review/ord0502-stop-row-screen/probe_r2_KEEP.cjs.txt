// 停靠行屏读探针 r2：上一趟（r1）停在 /v8/hub，只读到 today-heading，没进去行程屏 —— 那是我的走法错，
// 不是产品没有那一屏。路由词表（ROUTE_ALIASES）写着 itinerary/stays/wallet 都别名到 trips，
// 真地址是 /v8/trips；这一趟先直接进 /v8/trips，再逐枚量 trip-stop 的字、位置与中心命中。
const fs = require("fs");
const { chromium } = require("playwright");

const FRONT = process.env.STOPROW_FRONT;
const OUT = process.env.STOPROW_OUT;
const SHOT = process.env.STOPROW_SHOT;
const JOURNEY = process.env.STOPROW_JID;
const ANCHORS = ["trips-tabs", "trips-tab-itinerary", "trips-tab-bookings", "trips-tab-spend",
                 "trip-itinerary", "trip-itinerary-empty", "trip-stop", "trip-stop-details",
                 "trip-day-strip", "trip-week-strip", "trip-day-total", "stop-checkin",
                 "today-heading", "trial-setup", "hub-secondary-actions"];

const boxOf = (b) => (b ? { x: b.x, y: b.y, width: b.width, height: b.height,
                             bottom: b.y + b.height, right: b.x + b.width } : null);

(async () => {
  const browser = await chromium.launch();
  const ctx = await browser.newContext({
    viewport: { width: 390, height: 844 }, deviceScaleFactor: 3, isMobile: true,
    hasTouch: true, locale: "zh-CN", serviceWorkers: "block",
  });
  const page = await ctx.newPage();
  const rep = { front: FRONT, anchors_requested: ANCHORS, legs: [], rows: [], errors: [] };
  const snap = async (name, extra) => {
    const inv = {};
    for (const a of ANCHORS) {
      const loc = page.getByTestId(a);
      const n = await loc.count();
      inv[a] = { count: n,
                first_visible: n ? await loc.first().isVisible().catch(() => false) : false,
                first_text: n ? (await loc.first().innerText().catch(() => "")).replace(/\s+/g, " ").trim().slice(0, 80) : "" };
    }
    rep.legs.push(Object.assign({ leg: name, url: page.url(),
                                  body_text_chars: await page.evaluate(() => document.body.innerText.length).catch(() => -1) },
                                { inventory: inv }, extra || {}));
  };
  try {
    await page.addInitScript(([jid]) => {
      try { localStorage.setItem("citynexus.v8.journey_id", jid); } catch (e) {}
    }, [JOURNEY]);
    await page.goto(FRONT + "/v8/hub", { waitUntil: "domcontentloaded", timeout: 60000 });
    await page.waitForTimeout(3000);
    await snap("hub");
    await page.goto(FRONT + "/v8/trips", { waitUntil: "domcontentloaded", timeout: 60000 });
    await page.waitForTimeout(3000);
    await snap("trips_default_tab");
    const itab = page.getByTestId("trips-tab-itinerary");
    if (await itab.count()) {
      await itab.first().click({ timeout: 8000 }).catch((e) => rep.errors.push("tab " + e.message.slice(0, 70)));
      await page.waitForTimeout(2000);
      await snap("after_click_itinerary_tab");
    }
    const stops = page.getByTestId("trip-stop");
    const n = await stops.count();
    const tabsBox = boxOf(await page.getByTestId("trips-tabs").first().boundingBox().catch(() => null));
    for (let i = 0; i < n; i++) {
      const row = stops.nth(i);
      await row.scrollIntoViewIfNeeded().catch((e) => rep.errors.push("scroll " + i));
      await page.waitForTimeout(200);
      const b = boxOf(await row.boundingBox().catch(() => null));
      const text = (await row.innerText().catch(() => "")).replace(/\s+/g, " ").trim();
      let hit = null, overlap = null;
      if (b) {
        const cx = Math.round(b.x + b.width / 2), cy = Math.round(b.y + b.height / 2);
        hit = await page.evaluate(([x, y]) => {
          const el = document.elementFromPoint(x, y);
          if (!el) return { found: false };
          const anchor = (el.closest && el.closest("[data-testid]")) || el;
          return { found: true, testid: anchor.getAttribute ? anchor.getAttribute("data-testid") : null,
                   tag: el.tagName };
        }, [cx, cy]).catch((e) => ({ error: e.message.slice(0, 50) }));
        overlap = tabsBox ? (b.bottom > tabsBox.y && b.y < tabsBox.bottom) : null;
      }
      rep.rows.push({ index: i, text: text.slice(0, 180), box: b,
                      in_viewport: b ? (b.y >= 0 && b.bottom <= 844) : null,
                      center_hit: hit, overlaps_tabbar: overlap });
    }
    await page.screenshot({ path: SHOT }).catch((e) => rep.errors.push("shot"));
    rep.viewport = { width: 390, height: 844 };
    rep.row_count = rep.rows.length;
  } catch (e) {
    rep.errors.push("fatal " + (e && e.message ? e.message.slice(0, 160) : String(e)));
  } finally {
    fs.writeFileSync(OUT, JSON.stringify(rep, null, 2) + "\n", "utf8");
    await browser.close();
  }
  console.log("rows", rep.rows.length, "| legs", rep.legs.map((l) => l.leg + ":" + JSON.stringify(l.inventory["trip-stop"].count)).join(" "));
})();
