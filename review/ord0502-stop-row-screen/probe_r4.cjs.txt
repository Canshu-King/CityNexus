// 停靠行屏读探针 r4。r3 的两半之外补第三半，并修一处会被误读的空值：
//  A 静态一帧：页面不动逐枚量矩形。r3 里那些行中心点全在视口外（文档 y 从 862 起、视口只有 844 高），
//    elementsFromPoint 对视口外的点返回空数组 —— 枚枚 "stack": [] 说的是"点不在屏上"，不是"没有东西盖住"，
//    所以这里给每枚行加 center_in_viewport，把两件事分开写。
//  B 滚进视口（scrollIntoViewIfNeeded，只保证"进得来"）：量 boundingBox + 中心那一叠元素。
//  C 滚到"固定底栏之上"：先把 position:fixed 且横跨整宽的条量出来（r3 现读到 NAV.v8-mobile-nav，
//    rect y=768、h=76、z=10），再把每枚行的中心摆到那条带上沿之上重测一遍 —— 只有这一半能判
//    "那枚行到底能不能被点"，B 那半咬不住（B 的滚动量由浏览器决定，行可以停在被底栏压着的位置）。
const fs = require("fs");
const { chromium } = require("playwright");

const FRONT = process.env.STOPROW_FRONT;
const OUT = process.env.STOPROW_OUT;
const SHOT = process.env.STOPROW_SHOT;
const JOURNEY = process.env.STOPROW_JID;
const ANCHORS = ["trips-tabs", "trips-tab-itinerary", "trip-itinerary", "trip-stop",
                 "trip-stop-details", "trip-day-strip", "trip-day-total", "stop-checkin",
                 "today-heading"];

const boxOf = (b) => (b ? { x: b.x, y: b.y, width: b.width, height: b.height,
                            bottom: b.y + b.height, right: b.x + b.width } : null);

const MEASURE = () => {
  const sev = (el) => ({ tag: el.tagName, id: el.id || null,
    testid: el.getAttribute ? el.getAttribute("data-testid") : null,
    cls: (el.className && typeof el.className === "string") ? el.className.slice(0, 60) : null });
  const rows = Array.from(document.querySelectorAll('[data-testid="trip-stop"]')).map((el, i) => {
    const r = el.getBoundingClientRect();
    const cx = Math.round(r.x + r.width / 2), cy = Math.round(r.y + r.height / 2);
    const inView = cx >= 0 && cy >= 0 && cx <= window.innerWidth && cy <= window.innerHeight;
    const tp = inView ? document.elementFromPoint(cx, cy) : null;
    return { index: i,
             text: (el.innerText || "").replace(/\s+/g, " ").trim().slice(0, 180),
             rect: { x: r.x, y: r.y, width: r.width, height: r.height,
                    bottom: r.y + r.height, right: r.x + r.width },
             center: { x: cx, y: cy }, center_in_viewport: inView,
             top_of_stack: tp ? sev(tp) : null,
             stack: tp ? document.elementsFromPoint(cx, cy).slice(0, 5).map(sev) : [] };
  });
  const se = document.scrollingElement || document.documentElement;
  const fixed = Array.from(document.querySelectorAll("body *"))
    .filter((el) => getComputedStyle(el).position === "fixed")
    .map((el) => { const r = el.getBoundingClientRect();
      return Object.assign(sev(el), { rect: { x: r.x, y: r.y, width: r.width, height: r.height,
        bottom: r.y + r.height }, z: getComputedStyle(el).zIndex,
        spans_width: r.width >= window.innerWidth * 0.9 }); })
    .filter((e) => e.rect.width > 0 && e.rect.height > 0);
  return { rows: rows, fixed_elements: fixed,
           frame: { scroll_y: se.scrollTop, scroll_height: se.scrollHeight,
                    client_height: se.clientHeight, inner_width: window.innerWidth,
                    inner_height: window.innerHeight } };
};

// 把第 i 枚行的中心摆到 usable_top..bar_top 这条带的正中，然后回读它的矩形。
const CENTER_ABOVE_BAR = ([i, barTop]) => {
  const el = document.querySelectorAll('[data-testid="trip-stop"]')[i];
  if (!el) return null;
  const r = el.getBoundingClientRect();
  const target = barTop / 2;
  const dy = (r.y + r.height / 2) - target;
  window.scrollBy(0, dy);
  return { scrolled_by: dy };
};

(async () => {
  const browser = await chromium.launch();
  const ctx = await browser.newContext({
    viewport: { width: 390, height: 844 }, deviceScaleFactor: 3, isMobile: true,
    hasTouch: true, locale: "zh-CN", serviceWorkers: "block",
  });
  const page = await ctx.newPage();
  const rep = { front: FRONT, anchors_requested: ANCHORS, legs: [], static_frame: null,
                if_needed_rows: [], above_bar_rows: [], bar_band: null, errors: [] };
  const snap = async (name) => {
    const inv = {};
    for (const a of ANCHORS) {
      const loc = page.getByTestId(a);
      const n = await loc.count();
      inv[a] = { count: n,
                 first_visible: n ? await loc.first().isVisible().catch(() => false) : false };
    }
    rep.legs.push({ leg: name, url: page.url(),
                    body_text_chars: await page.evaluate(() => document.body.innerText.length)
                      .catch(() => -1), inventory: inv });
  };
  try {
    await page.addInitScript(([jid]) => {
      try { localStorage.setItem("citynexus.v8.journey_id", jid); } catch (e) {}
    }, [JOURNEY]);
    await page.goto(FRONT + "/v8/trips", { waitUntil: "domcontentloaded", timeout: 60000 });
    await page.waitForTimeout(3500);
    const itab = page.getByTestId("trips-tab-itinerary");
    if (await itab.count()) {
      await itab.first().click({ timeout: 8000 })
        .catch((e) => rep.errors.push("tab " + e.message.slice(0, 60)));
      await page.waitForTimeout(1500);
    }
    await snap("trips_itinerary_default");
    rep.static_frame = await page.evaluate(MEASURE);
    // 底栏带：横跨整宽、且位置在视口下半的 fixed 元素，取其上沿
    const bars = (rep.static_frame.fixed_elements || [])
      .filter((e) => e.spans_width && e.rect.y > 400);
    rep.bar_band = bars.length ? { count: bars.length, top: Math.min.apply(null, bars.map((e) => e.rect.y)),
                                   items: bars } : { count: 0, top: null, items: [] };
    const usableTop = rep.bar_band.count ? rep.bar_band.top : rep.static_frame.frame.inner_height;
    const n = (rep.static_frame.rows || []).length;

    for (let i = 0; i < n; i++) {          // B 半：只保证滚进视口
      const row = page.getByTestId("trip-stop").nth(i);
      await row.scrollIntoViewIfNeeded().catch((e) => rep.errors.push("scrollB " + i));
      await page.waitForTimeout(150);
      const m = await page.evaluate(MEASURE);
      const r = m.rows[i] || {};
      rep.if_needed_rows.push({ index: i, scroll_y: m.frame.scroll_y, rect: r.rect,
        center_in_viewport: r.center_in_viewport, top_of_stack: r.top_of_stack,
        clears_bar: r.rect ? r.rect.bottom <= usableTop : null,
        under_bar: r.rect ? (r.rect.bottom > rep.bar_band.top && r.rect.y < rep.bar_band.top) : null,
        text: r.text });
    }
    for (let i = 0; i < n; i++) {          // C 半：摆到底栏之上再测
      await page.evaluate(CENTER_ABOVE_BAR, [i, usableTop]);
      await page.waitForTimeout(150);
      const m = await page.evaluate(MEASURE);
      const r = m.rows[i] || {};
      rep.above_bar_rows.push({ index: i, scroll_y: m.frame.scroll_y, rect: r.rect,
        center_in_viewport: r.center_in_viewport, top_of_stack: r.top_of_stack,
        stack: r.stack, clears_bar: r.rect ? r.rect.bottom <= usableTop : null,
        text: r.text });
    }
    await page.screenshot({ path: SHOT }).catch(() => rep.errors.push("shot"));
    rep.viewport = { width: 390, height: 844 };
    rep.counts = { static: n, if_needed: rep.if_needed_rows.length,
                   above_bar: rep.above_bar_rows.length };
  } catch (e) {
    rep.errors.push("fatal " + (e && e.message ? e.message.slice(0, 160) : String(e)));
  } finally {
    fs.writeFileSync(OUT, JSON.stringify(rep, null, 2) + "\n", "utf8");
    await browser.close();
  }
  console.log("rows", JSON.stringify(rep.counts), "| bar", JSON.stringify(rep.bar_band &&
              { count: rep.bar_band.count, top: rep.bar_band.top,
                first: rep.bar_band.items && rep.bar_band.items[0] && rep.bar_band.items[0].cls }),
              "| frame", JSON.stringify(rep.static_frame && rep.static_frame.frame),
              "| under_bar", rep.if_needed_rows.filter((r) => r.under_bar).length,
              "| above_bar_clears", rep.above_bar_rows.filter((r) => r.clears_bar).length);
})();
