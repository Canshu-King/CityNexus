// 停靠行屏读探针：在手机竖屏形状下，把每一枚 `trip-stop` 的字、位置、中心命中与底栏重叠都量出来。
// 只读自己车道的私有口；断言读的是渲染文字，不是 data 属性。
const path = require("path");
const fs = require("fs");
const { chromium } = require("playwright");

const FRONT = process.env.STOPROW_FRONT;
const OUT = process.env.STOPROW_OUT;
const SHOT = process.env.STOPROW_SHOT;
const JOURNEY = process.env.STOPROW_JID;
const ANCHORS = ["trip-itinerary", "trip-stop", "trip-stop-details", "trips-tabs",
                 "trip-day-strip", "trip-week-strip", "trip-day-total", "calendar-export",
                 "stop-checkin", "today-heading", "trial-setup"];

function boxOf(el) {
  const b = el ? el : null;
  if (!b) return null;
  return { x: b.x, y: b.y, width: b.width, height: b.height,
           bottom: b.y + b.height, right: b.x + b.width };
}

(async () => {
  const browser = await chromium.launch({ args: ["--disable-service-worker"] });
  const ctx = await browser.newContext({
    viewport: { width: 390, height: 844 }, deviceScaleFactor: 3, isMobile: true,
    hasTouch: true, locale: "zh-CN", serviceWorkers: "block",
  });
  const page = await ctx.newPage();
  const report = { front: FRONT, journey_id_set: JOURNEY, anchors_requested: ANCHORS,
                   steps: [], rows: [], errors: [], inventory: {} };
  try {
    await page.addInitScript(([jid]) => {
      try { localStorage.setItem("citynexus.v8.journey_id", jid); } catch (e) {}
    }, [JOURNEY]);
    await page.goto(FRONT + "/v8/hub", { waitUntil: "domcontentloaded", timeout: 60000 });
    await page.waitForTimeout(3000);
    report.steps.push({ where: "hub", url: page.url,
                        title: await page.title(),
                        heading_present: await page.getByTestId("today-heading").count() });

    // 行程页签（源里 data-testid="trips-tabs" 那一族按钮）
    const tabs = page.locator('[data-testid="trips-tabs"] button, [data-testid="trips-tabs"] [role="tab"]');
    const tabCount = await tabs.count();
    const tabLabels = [];
    for (let i = 0; i < tabCount; i++) {
      tabLabels.push((await tabs.nth(i).innerText()).replace(/\s+/g, " ").trim());
    }
    report.tabs = { count: tabCount, labels: tabLabels };
    let clicked = null;
    for (let i = 0; i < tabCount; i++) {
      const label = tabLabels[i] || "";
      if (label.indexOf("行程") >= 0 || /itinerary/i.test(label)) {
        await tabs.nth(i).click({ timeout: 8000 }).catch((e) => report.errors.push("tab click " + e.message.slice(0, 80)));
        clicked = label;
        break;
      }
    }
    report.tab_clicked = clicked;
    await page.waitForTimeout(2500);

    for (const a of ANCHORS) {
      const loc = page.getByTestId(a);
      const n = await loc.count();
      const firstVisible = n ? await loc.first().isVisible().catch(() => false) : false;
      const firstText = n ? (await loc.first().innerText().catch(() => "")).replace(/\s+/g, " ").trim().slice(0, 90) : "";
      report.inventory[a] = { count: n, first_visible: firstVisible, first_text: firstText };
    }

    const stops = page.getByTestId("trip-stop");
    const count = await stops.count();
    const tabbarBox = boxOf(await page.locator('[data-testid="trips-tabs"]').boundingBox().catch(() => null));
    for (let i = 0; i < count; i++) {
      const row = stops.nth(i);
      await row.scrollIntoViewIfNeeded().catch((e) => report.errors.push("scroll " + i + " " + e.message.slice(0, 60)));
      await page.waitForTimeout(250);
      const b = boxOf(await row.boundingBox().catch(() => null));
      const text = (await row.innerText().catch(() => "")).replace(/\s+/g, " ").trim();
      let hit = null, overlapped = null;
      if (b) {
        const cx = Math.round(b.x + b.width / 2), cy = Math.round(b.y + b.height / 2);
        hit = await page.evaluate(([x, y]) => {
          const el = document.elementFromPoint(x, y);
          if (!el) return { found: false };
          const rec = el.closest("[data-testid]") || el;
          return { found: true, testid: rec.getAttribute ? rec.getAttribute("data-testid") : null,
                   tag: el.tagName, cls: (el.className || "").toString().slice(0, 60) };
        }, [cx, cy]).catch((e) => ({ error: e.message.slice(0, 60) }));
        overlapped = tabbarBox ? (b.bottom > tabbarBox.y && b.y < tabbarBox.bottom) : null;
      }
      report.rows.push({ index: i, text: text.slice(0, 160), box: b,
                         in_viewport: b ? (b.y >= 0 && b.bottom <= 844) : null,
                         center_hit: hit, overlaps_tabbar: overlapped });
    }
    await page.screenshot({ path: SHOT, fullPage: false }).catch((e) => report.errors.push("shot " + e.message.slice(0, 60)));
    report.body_text_chars = (await page.evaluate(() => document.body.innerText.length).catch(() => -1));
    report.viewport = { width: 390, height: 844 };
  } catch (e) {
    report.errors.push("fatal " + (e && e.message ? e.message.slice(0, 160) : String(e)));
  } finally {
    fs.writeFileSync(OUT, JSON.stringify(report, null, 2) + "\n", "utf8");
    await browser.close();
  }
  console.log("rows", report.rows.length, "| stops anchor", JSON.stringify(report.inventory["trip-stop"]), "| errors", report.errors.length);
})();
