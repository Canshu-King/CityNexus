# -*- coding: utf-8 -*-
"""按现读尖构建前端产物（为停靠行屏读做准备）。
只在**自己的**快照目录里做：`git archive <尖>` 导出 → 用 junction 借用同锁依赖 → vite 构建。
依赖是否同一锁要量，不假设：比对 `frontend/package.json` 与 `frontend/package-lock.json` 两枚 blob。
junction 的存在与拆除都记在回执里（拆除用 rmdir，绝不递归删——递归删会顺着 junction 吃掉别人的 node_modules）。"""
from __future__ import annotations

import hashlib
import json
import os
import shutil
import subprocess
import sys
import time
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
CHECK = ("backend/app/main.py", "backend/app/demo_scene.py", "frontend/src/features/trip/TripItinerary.tsx",
         "frontend/src/app/V8Experience.tsx", "frontend/package.json", "frontend/package-lock.json",
         "frontend/vite.config.ts", "data/mock/places.json")
OUT = ART / "stoprow_build_state_r1.json"


def g(*a: str, cwd: Path = WT2) -> str:
    r = subprocess.run(["git", *a], cwd=str(cwd), capture_output=True, text=True,
                       encoding="utf-8", errors="replace")
    return (r.stdout or "").strip()


def sha1(p: Path) -> str:
    return hashlib.sha1(p.read_bytes()).hexdigest()


def blob_id(p: Path) -> str:
    return g("hash-object", "--no-filters", str(p), cwd=WT2)


def main() -> int:
    tip = g("ls-remote", "--refs", "origin", MIRROR).split("\t")[0]
    if len(tip) != 40:
        print("REFUSED: 读不到尖")
        return 1
    g("fetch", "-q", "origin", tip)
    run_id = "STOPROW-" + time.strftime("%H%M%S")
    snap = Path("D:/moodle") / ("_qoder_stoprow_%s" % run_id)
    snap.mkdir(parents=True, exist_ok=True)
    proc = subprocess.run(["git", "archive", "--format=tar", tip], cwd=str(WT2), capture_output=True)
    if proc.returncode != 0:
        print("REFUSED: archive rc=%d" % proc.returncode)
        return 1
    tar = snap / "a.tar"
    tar.write_bytes(proc.stdout)
    import tarfile
    with tarfile.open(str(tar)) as t:
        t.extractall(path=str(snap), filter="data")
    tar.unlink()
    blobs = {}
    for rel in CHECK:
        want = g("rev-parse", "%s:%s" % (tip, rel))
        have = blob_id(snap / rel) if (snap / rel).exists() else ""
        blobs[rel] = {"tip_blob": want, "disk_blob": have, "equal": bool(want) and want == have}
    if not all(v["equal"] for v in blobs.values()):
        bad = [k for k, v in blobs.items() if not v["equal"]]
        print("REFUSED: 导出的字节与尖不等的有：", bad)
        return 1
    # 同一把锁才谈得上"借用依赖"
    same_lock = (g("rev-parse", "%s:frontend/package-lock.json" % tip)
                 == g("rev-parse", "HEAD:frontend/package-lock.json"))
    same_pkg = (g("rev-parse", "%s:frontend/package.json" % tip)
                == g("rev-parse", "HEAD:frontend/package.json"))
    link = snap / "frontend" / "node_modules"
    junc = None
    if same_lock and same_pkg:
        r = subprocess.run(["cmd", "/c", "mklink", "/J", str(link),
                            str(WT2 / "frontend" / "node_modules")],
                           capture_output=True, text=True)
        junc = {"rc": r.returncode, "made": link.is_dir(), "target": str(WT2 / "frontend" / "node_modules")}
        if not (r.returncode == 0 and link.is_dir()):
            print("REFUSED: junction 没做成：", r.stdout[:120], r.stderr[:120])
            return 1
    else:
        print("REFUSED: 依赖锁不同代，不借用别人的 node_modules")
        return 1

    vite_js = WT2 / "frontend/node_modules/vite/bin/vite.js"
    started = time.time()
    node = shutil.which("node") or "node"
    build = subprocess.run([node, str(vite_js), "build", "--logLevel", "warn"],
                           cwd=str(snap / "frontend"), capture_output=True, text=True,
                           encoding="utf-8", errors="replace", timeout=900)
    secs = round(time.time() - started, 1)
    dist = snap / "frontend" / "dist"
    assets = sorted(p for p in (dist / "assets").glob("*")) if (dist / "assets").exists() else []
    listing = [{"name": p.name, "bytes": p.stat().st_size, "sha1": sha1(p)} for p in assets]
    index_html = (dist / "index.html").read_text(encoding="utf-8", errors="replace") \
        if (dist / "index.html").exists() else ""
    entry = [l for l in listing if l["name"].startswith("index-") and l["name"].endswith(".js")]
    state = {"run_id": run_id, "line_tip": tip, "snap": str(snap), "build_rc": build.returncode,
             "build_seconds": secs,
             "build_tail": (build.stdout or "")[-500:] + "|" + (build.stderr or "")[-300:],
             "deps_same_lock": {"package.json": same_pkg, "package-lock.json": same_lock},
             "junction": junc, "blobs": blobs,
             "asset_count": len(listing), "assets": listing,
             "index_entry_script": [x for x in re_find(index_html)]}
    OUT.write_text(json.dumps(state, ensure_ascii=False, indent=2) + "\n", encoding="utf-8",
                   newline=chr(10))
    print(json.dumps({k: state[k] for k in ("run_id", "line_tip", "snap", "build_rc",
                                            "build_seconds", "deps_same_lock", "junction",
                                            "asset_count", "index_entry_script")},
                     ensure_ascii=False, indent=2))
    print("入口 JS:", json.dumps(entry, ensure_ascii=False))
    print("STATE", OUT)
    return 1 if build.returncode != 0 else 0


def re_find(html: str):
    import re
    for m in re.finditer(r'<script[^>]*src="([^"]+)"', html):
        yield m.group(1)
    for m in re.finditer(r'<link[^>]*rel="stylesheet"[^>]*href="([^"]+)"', html):
        yield m.group(1)


if __name__ == "__main__":
    sys.exit(main())
