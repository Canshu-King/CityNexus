"""停靠行屏读这一腿：自己的私有后端（供出场景）+ 自己按尖构建的产物 + 手机竖屏 Chromium。
三件都要证明：
 ① 屏上那一代 = 磁盘上那一代：入口资产「被服务字节」与「磁盘字节」逐字节比（本机 sha1，不是 blob）；
 ② 停靠行的字与位置：每枚 `trip-stop` 的渲染文字、boundingBox、中心 elementFromPoint 命中、与页签底栏是否重叠；
 ③ 屏上的名字与答复、与磁盘上的封存夹具对得上（三处同名才算数）。"""
from __future__ import annotations

import hashlib
import json
import os
import socket
import subprocess
import sys
import threading
import time
import urllib.error
import urllib.request
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
SCR = Path("D:/moodle/_qoder_newlane_ord0444_20260929/scratch")
WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
BUILD = json.loads((ART / "stoprow_build_state_r1.json").read_text(encoding="utf-8"))
SNAP = Path(BUILD["snap"])
DIST = SNAP / "frontend" / "dist"
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
FORBIDDEN = {8000, 4781, 4782, 4783, 4785, 4787, 4789, 8693, 4799}
BAD = []


def sha1_bytes(b: bytes) -> str:
    return hashlib.sha1(b).hexdigest()


def free_port(wanted: int, taken=()) -> int:
    """要一枚真能 bind、且不在已占用集合里的号（上一版两次 free_port 抓到同一枚号，
    后端 bind 失败、代理把自己当上游 ⇒ 502）。"""

    p = wanted
    while p < wanted + 60:
        if p not in FORBIDDEN and p not in taken:
            with socket.socket() as s:
                try:
                    s.bind(("127.0.0.1", p))
                    return p
                except OSError:
                    pass
        p += 1
    raise SystemExit("找不到空闲私有端口")


def http(url, data=None, headers=None, timeout=60):
    req = urllib.request.Request(url, data=data, method="POST" if data is not None else "GET",
                                 headers=headers or {})
    try:
        with urllib.request.urlopen(req, timeout=timeout) as r:
            return r.status, r.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()
    except Exception as e:  # noqa: BLE001
        return None, (type(e).__name__ + str(e))[:160].encode()


def main() -> int:
    tip = subprocess.run(["git", "ls-remote", "--refs", "origin", MIRROR], cwd=str(WT2),
                         capture_output=True, text=True).stdout.split("\t")[0].strip()
    # 每一趟一枚新文件名：那扇 `_refuse_to_reopen_an_existing_e2e_database` 会拒绝复用同一 run 的库，
    # 而重建 run_id 才是它给的正解（不许在这儿删库）。
    run_id = "%s-w%s" % (BUILD["run_id"], time.strftime("%H%M%S"))
    db = SNAP / "backend" / ("%s.db" % run_id)
    api = free_port(4833)
    front = free_port(api + 3, taken={api})
    env = dict(os.environ)
    for k in ("HTTP_PROXY", "HTTPS_PROXY", "http_proxy", "https_proxy"):
        env.pop(k, None)
    env["CITYNEXUS_DATABASE_URL"] = "sqlite:///%s" % db.as_posix()
    env["CITYNEXUS_E2E_RUN_ID"] = run_id
    env["CITYNEXUS_BACKEND_PORT"] = str(api)
    env["CITYNEXUS_DEMO_MODE"] = "true"
    token = os.urandom(16).hex()
    env["CITYNEXUS_DEMO_SCENE_TOKEN"] = token
    out = {"run_id": run_id, "line_tip_at_walk": tip, "build_tip": BUILD["line_tip"],
           "snap": str(SNAP), "api_port": api, "front_port": front,
           "dist_assets": BUILD["assets"], "token_names_only": ["CITYNEXUS_DEMO_SCENE_TOKEN"]}
    if tip != BUILD["line_tip"]:
        BAD.append("走查时尖已换代（%s → %s）：块里的代次必须按这一枚写" % (BUILD["line_tip"][:8], tip[:8]))
        out["generation_note"] = "构建与走查之间线换了一代"
    logf = ART / ("uvicorn_%s_walk.log" % run_id)
    backend = subprocess.Popen([str(Path("D:/moodle/CityNexus/.venv/Scripts/python.exe")),
                                "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1",
                                "--port", str(api)], cwd=str(SNAP / "backend"), env=env,
                               stdout=logf.open("w", encoding="utf-8"), stderr=subprocess.STDOUT)
    out["backend_pid"] = backend.pid
    served = {}

    class H(BaseHTTPRequestHandler):
        def log_message(self, *a):
            pass

        def _send(self, code, body, ctype):
            self.send_response(code)
            self.send_header("Content-Type", ctype)
            self.send_header("Content-Length", str(len(body)))
            self.send_header("Cache-Control", "no-store")
            self.end_headers()
            self.wfile.write(body)

        def do_GET(self):
            p = self.path.split("?")[0]
            if p.startswith("/api/"):
                st, body = http("http://127.0.0.1:%d%s" % (api, self.path))
                self._send(st or 502, body, "application/json")
                return
            rel = p.lstrip("/") or "index.html"
            f = (DIST / rel).resolve()
            if not str(f).startswith(str(DIST)) or not f.is_file():
                f = DIST / "index.html"
            body = f.read_bytes()
            ctype = ("text/javascript" if f.suffix == ".js" else "text/css" if f.suffix == ".css"
                     else "text/html" if f.suffix == ".html" else "application/octet-stream")
            self._send(200, body, ctype)

        do_POST = do_GET

    srv = ThreadingHTTPServer(("127.0.0.1", front), H)
    th = threading.Thread(target=srv.serve_forever, daemon=True)
    th.start()
    try:
        for _ in range(90):
            st, body = http("http://127.0.0.1:%d/api/v1/health" % api)
            if st == 200 and run_id.encode() in body:
                break
            time.sleep(1)
        else:
            BAD.append("后端没答或答复里没带本 run id")
            out["BAD"] = BAD
            (ART / ("stoprow_walk_failure_%s.json" % run_id)).write_text(
                json.dumps(out, ensure_ascii=False, indent=2) + "
", encoding="utf-8",
                newline=chr(10))
            print("REFUSED:", *BAD, sep="
  ")
            print("LOG", (ART / ("uvicorn_%s_walk.log" % run_id)).name)
            raise SystemExit(1)
        out["health_via_api"] = {"http_status": st, "carries_run_id": run_id in body.decode("utf-8", "replace")}
        st, body = http("http://127.0.0.1:%d/api/v1/health" % front)
        out["health_via_front_origin"] = {"http_status": st,
                                          "carries_run_id": run_id in body.decode("utf-8", "replace")}
        if not out["health_via_front_origin"]["carries_run_id"]:
            BAD.append("经前端原点没证到归属（代理指向别处）")

        hdr = {"Content-Type": "application/json", "X-Demo-Scene-Token": token,
               "Idempotency-Key": "walk-" + run_id}
        req = urllib.request.Request("http://127.0.0.1:%d/api/v1/demo/scene/ensure" % api,
                                      data=b"{}", headers=hdr, method="POST")
        with urllib.request.urlopen(req, timeout=120) as r:
            rec = json.loads(r.read().decode("utf-8"))
        jid = rec.get("journey_id")
        out["ensure"] = {"http_status": r.status, "state": rec.get("state"),
                         "journey_status": rec.get("journey_status")}
        if not jid or rec.get("state") != "READY":
            BAD.append("供不出场景，屏读没有前置")
            return 1
        st, body = http("http://127.0.0.1:%d/api/v1/journeys/%s/plans" % (front, jid))
        plans = json.loads(body.decode("utf-8"))
        sel = next((p for p in plans if p.get("status") == "SELECTED"), None)
        acts = (sel or {}).get("activities") or []
        out["api_stop_titles"] = [a.get("title") for a in acts]
        out["api_stop_count"] = len(acts)

        # ① 字节对照：入口资产被服务的那份 == 磁盘那份
        for a in BUILD["assets"]:
            if a["name"].startswith("index-") and a["name"].endswith((".js", ".css")):
                st, body = http("http://127.0.0.1:%d/assets/%s" % (front, a["name"]))
                disk = (DIST / "assets" / a["name"]).read_bytes()
                served[a["name"]] = {"http_status": st, "served_sha1": sha1_bytes(body),
                                     "disk_sha1": sha1_bytes(disk), "bytes": len(body),
                                     "served_equals_disk": body == disk,
                                     "equals_build_record": sha1_bytes(disk) == a["sha1"]}
        out["byte_pairing"] = served
        if not served or not all(v["served_equals_disk"] and v["equals_build_record"]
                                 for v in served.values()):
            BAD.append("被服务字节与磁盘/构建记录不逐字等 —— 屏读不能归到这一代")
        fx = (SNAP / "data" / "mock" / "places.json")
        out["fixture"] = {"path": "data/mock/places.json", "bytes": fx.stat().st_size,
                          "disk_sha1": sha1_bytes(fx.read_bytes())}

        shot = ART / ("stoprow_screen_%s.png" % run_id)
        dest = ART / ("stoprow_screen_%s.json" % run_id)
        penv = dict(os.environ)
        penv.update({"NODE_PATH": str(WT2 / "frontend" / "node_modules"),
                     "STOPROW_FRONT": "http://127.0.0.1:%d" % front,
                     "STOPROW_OUT": str(dest), "STOPROW_SHOT": str(shot),
                     "STOPROW_JID": jid})
        node = subprocess.run([str(WT2 / "frontend" / "node_modules" / ".bin" / "node")
                               if (WT2 / "frontend" / "node_modules" / ".bin" / "node").exists()
                               else "node", str(SCR / "stoprow_probe_r1.cjs")],
                              capture_output=True, text=True, encoding="utf-8", errors="replace",
                              env=penv, timeout=420)
        out["probe_rc"] = node.returncode
        out["probe_stdout"] = (node.stdout or "")[-400:]
        out["probe_stderr"] = (node.stderr or "")[-400:]
        rep = json.loads(dest.read_text(encoding="utf-8")) if dest.exists() else {}
        out["screen"] = {"url_after_load": rep.get("steps"), "tabs": rep.get("tabs"),
                         "tab_clicked": rep.get("tab_clicked"),
                         "inventory": rep.get("inventory"),
                         "rows": rep.get("rows"), "body_text_chars": rep.get("body_text_chars"),
                         "errors": rep.get("errors"), "viewport": rep.get("viewport")}
        names = [r["text"].split("\n")[0].strip() for r in (rep.get("rows") or [])]
        out["screen_stop_texts_first_line"] = [r["text"][:60] for r in (rep.get("rows") or [])]
        clean = [n.split("  ")[0].strip() for n in names]
        out["name_triple_check"] = {
            "api_titles": out["api_stop_titles"],
            "screen_rows": len(rep.get("rows") or []),
            "screen_first_lines": names[:8],
            "every_row_has_a_box": all(r.get("box") for r in (rep.get("rows") or [])),
            "every_center_hits_a_testid_anchor": all(
                ((r.get("center_hit") or {}).get("testid")) for r in (rep.get("rows") or [])),
            "any_row_overlaps_tabbar": any(r.get("overlaps_tabbar") for r in (rep.get("rows") or [])),
        }
        fixture_text = fx.read_text(encoding="utf-8", errors="replace")
        out["name_triple_check"]["names_present_in_fixture"] = [
            {"title": t, "in_fixture": t in fixture_text,
             "on_screen": any(t and t in (r.get("text") or "") for r in (rep.get("rows") or []))}
            for t in (out["api_stop_titles"] or [])]
        if not (rep.get("rows")):
            BAD.append("屏上一枚停靠行都没读到 —— 这一格不能写成已交")
    finally:
        srv.shutdown()
        srv.server_close()
        backend.terminate()
        try:
            backend.wait(timeout=40)
        except subprocess.TimeoutExpired:
            backend.kill()
        out["backend_exit"] = backend.poll()
    out["BAD"] = BAD
    dst = ART / ("stoprow_walk_%s.json" % run_id)
    dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + "\n", encoding="utf-8",
                   newline=chr(10))
    print(json.dumps({k: out[k] for k in ("line_tip_at_walk", "build_tip", "ensure",
                                          "api_stop_count", "byte_pairing", "probe_rc",
                                          "screen")}, ensure_ascii=False, indent=2)[:3000])
    print("ARTIFACT", dst)
    print("BAD", BAD)
    return 1 if BAD else 0


if __name__ == "__main__":
    sys.exit(main())
