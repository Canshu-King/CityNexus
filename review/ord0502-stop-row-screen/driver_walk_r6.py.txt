# -*- coding: utf-8 -*-
"""停靠行屏读 r6 = r5 同一把尺（r5 那一趟用的是 stoprow_probe_r3.cjs 的旧回执、被形制闸拦下，
所以这条才是带 A/B/C 三半读数的趟）。换 stoprow_probe_r4.cjs：同一屏分三半量 ——
 A 静态一帧（页面不动）逐枚 trip-stop 的矩形 +「中心是否在视口内」（r3 里那些空 stack 说的是"点不在屏上"，
   不是"没有东西盖住"，这一版把两件事拆开）；
 B 只滚进视口（scrollIntoViewIfNeeded）后的矩形 + 中心那一叠元素 + 是否被固定底栏压住；
 C 把行中心摆到底栏上沿之上后的矩形 + 命中 + 是否真的越过底栏。
只有 C 能判「那枚行到底点得到点不到」；B 的滚动量由浏览器决定，行可以停在被底栏盖着的位置。
另外两条腿不变：① 入口资产「被服务字节 == 磁盘字节 == 构建记录字节」逐字比（本机 sha1，不是 git blob）；
③ 屏上的字 / 答复里的 title / 磁盘封存夹具里的名字，三处同名才算数。
每一趟一枚新库文件名：那扇 _refuse_to_reopen_an_existing_e2e_database 拒复用，正解是换新 run（不删库）。"""
from __future__ import annotations

import hashlib
import json
import os
import socket
import subprocess
import sys
import threading
import time
import urllib.error
import urllib.request
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
SCR = Path("D:/moodle/_qoder_newlane_ord0444_20260929/scratch")
WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
BUILD = json.loads((ART / "stoprow_build_state_r1.json").read_text(encoding="utf-8"))
SNAP = Path(BUILD["snap"])
DIST = SNAP / "frontend" / "dist"
PROBE = SCR / "stoprow_probe_r4.cjs"
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
PYX = Path("D:/moodle/CityNexus/.venv/Scripts/python.exe")
FORBIDDEN = {8000, 4781, 4782, 4783, 4785, 4787, 4789, 8693, 4799}
BAD: list[str] = []


def sha1_bytes(b: bytes) -> str:
    return hashlib.sha1(b).hexdigest()


def free_port(wanted: int, taken=()) -> int:
    p = wanted
    while p < wanted + 60:
        if p not in FORBIDDEN and p not in taken:
            with socket.socket() as s:
                try:
                    s.bind(("127.0.0.1", p))
                    return p
                except OSError:
                    pass
        p += 1
    raise SystemExit("找不到空闲私有端口")


def http(url, data=None, headers=None, timeout=60):
    req = urllib.request.Request(url, data=data, method="POST" if data is not None else "GET",
                                 headers=headers or {})
    try:
        with urllib.request.urlopen(req, timeout=timeout) as r:
            return r.status, r.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()
    except Exception as e:  # noqa: BLE001
        return None, (type(e).__name__ + str(e))[:160].encode()


def main() -> int:
    tip = subprocess.run(["git", "ls-remote", "--refs", "origin", MIRROR], cwd=str(WT2),
                         capture_output=True, text=True).stdout.split("\t")[0].strip()
    run_id = "%s-w%s" % (BUILD["run_id"], time.strftime("%H%M%S"))
    db = SNAP / "backend" / ("%s.db" % run_id)
    api = free_port(4833)
    front = free_port(api + 3, taken={api})
    env = dict(os.environ)
    for k in ("HTTP_PROXY", "HTTPS_PROXY", "http_proxy", "https_proxy"):
        env.pop(k, None)
    env["CITYNEXUS_DATABASE_URL"] = "sqlite:///%s" % db.as_posix()
    env["CITYNEXUS_E2E_RUN_ID"] = run_id
    env["CITYNEXUS_BACKEND_PORT"] = str(api)
    env["CITYNEXUS_DEMO_MODE"] = "true"
    token = os.urandom(16).hex()
    env["CITYNEXUS_DEMO_SCENE_TOKEN"] = token
    out = {"run_id": run_id, "probe": PROBE.name,
           "driver": Path(__file__).name, "line_tip_at_walk": tip,
           "build_tip": BUILD["line_tip"], "snap": str(SNAP), "api_port": api,
           "front_port": front,
           "token_names_only": ["CITYNEXUS_DEMO_SCENE_TOKEN", "X-Demo-Scene-Token"]}
    if tip != BUILD["line_tip"]:
        out["generation_note"] = "构建那趟尖 %s，走查这趟现读 %s" % (BUILD["line_tip"], tip)

    log_path = ART / ("uvicorn_%s_walk.log" % run_id)
    backend = subprocess.Popen([str(PYX), "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1",
                                "--port", str(api)], cwd=str(SNAP / "backend"), env=env,
                               stdout=log_path.open("w", encoding="utf-8"),
                               stderr=subprocess.STDOUT)
    out["backend_pid"] = backend.pid
    served = {}

    class H(BaseHTTPRequestHandler):
        def log_message(self, *a):
            pass

        def _send(self, code, body, ctype):
            self.send_response(code)
            self.send_header("Content-Type", ctype)
            self.send_header("Content-Length", str(len(body)))
            self.send_header("Cache-Control", "no-store")
            self.end_headers()
            try:
                self.wfile.write(body)
            except (ConnectionAbortedError, ConnectionResetError, BrokenPipeError):
                pass

        def do_GET(self):
            p = self.path.split("?")[0]
            if p.startswith("/api/"):
                st, body = http("http://127.0.0.1:%d%s" % (api, self.path))
                self._send(st or 502, body, "application/json")
                return
            rel = (p.lstrip("/") or "index.html")
            f = (DIST / rel).resolve()
            ctype = ("text/javascript" if f.suffix == ".js" else "text/css" if f.suffix == ".css"
                     else "text/html" if f.suffix == ".html" else "application/octet-stream")
            if not str(f).startswith(str(DIST)) or not f.is_file():
                f, ctype = DIST / "index.html", "text/html"
            self._send(200, f.read_bytes(), ctype)

        do_POST = do_GET

    srv = ThreadingHTTPServer(("127.0.0.1", front), H)
    threading.Thread(target=srv.serve_forever, daemon=True).start()
    try:
        ok, st, body = False, None, b""
        for _ in range(90):
            st, body = http("http://127.0.0.1:%d/api/v1/health" % api)
            if st == 200 and run_id.encode() in body:
                ok = True
                break
            time.sleep(1)
        out["health_via_api"] = {"ok": ok, "http_status": st,
                                 "carries_run_id": run_id in body.decode("utf-8", "replace")}
        if not ok:
            BAD.append("后端没答或答复里没本 run id（日志 %s）" % log_path.name)
            raise SystemExit(1)
        st, body = http("http://127.0.0.1:%d/api/v1/health" % front)
        out["health_via_front_origin"] = {"http_status": st,
                                          "carries_run_id": run_id in body.decode("utf-8", "replace")}
        if not out["health_via_front_origin"]["carries_run_id"]:
            BAD.append("经前端原点没证到归属（代理指向别处）")

        hdr = {"Content-Type": "application/json", "X-Demo-Scene-Token": token,
               "Idempotency-Key": "walk-" + run_id}
        req = urllib.request.Request("http://127.0.0.1:%d/api/v1/demo/scene/ensure" % api,
                                     data=b"{}", headers=hdr, method="POST")
        with urllib.request.urlopen(req, timeout=120) as r:
            rec = json.loads(r.read().decode("utf-8"))
        jid = rec.get("journey_id")
        out["ensure"] = {"http_status": r.status, "state": rec.get("state"),
                         "journey_status": rec.get("journey_status")}
        if not jid or rec.get("state") != "READY":
            BAD.append("供不出场景，屏读没有前置")
            raise SystemExit(1)
        out["journey_id_present"] = bool(jid)
        st, body = http("http://127.0.0.1:%d/api/v1/journeys/%s/plans" % (front, jid))
        plans = json.loads(body.decode("utf-8"))
        sel = next((p for p in plans if p.get("status") == "SELECTED"), None)
        acts = (sel or {}).get("activities") or []
        out["api_stop_count"] = len(acts)
        out["api_stop_titles"] = [a.get("title") for a in acts]
        out["api_stop_place_ids"] = [a.get("place_id") for a in acts]

        for a in BUILD["assets"]:
            if a["name"].startswith("index-") and a["name"].endswith((".js", ".css")):
                st, body = http("http://127.0.0.1:%d/assets/%s" % (front, a["name"]))
                disk = (DIST / "assets" / a["name"]).read_bytes()
                served[a["name"]] = {"http_status": st, "bytes": len(body),
                                     "served_sha1": sha1_bytes(body), "disk_sha1": sha1_bytes(disk),
                                     "served_equals_disk": body == disk,
                                     "disk_equals_build_record": sha1_bytes(disk) == a["sha1"]}
        out["byte_pairing"] = served
        if not served or not all(v["served_equals_disk"] and v["disk_equals_build_record"]
                                 for v in served.values()):
            BAD.append("被服务字节与磁盘/构建记录不逐字等 —— 屏读不能归到这一代")
        fx = SNAP / "data" / "mock" / "places.json"
        out["fixture"] = {"rel": "data/mock/places.json", "bytes": fx.stat().st_size,
                          "disk_sha1": sha1_bytes(fx.read_bytes())}

        shot = ART / ("stoprow_screen_%s.png" % run_id)
        dest = ART / ("stoprow_screen_%s.json" % run_id)
        penv = dict(os.environ)
        penv.update({"NODE_PATH": str(WT2 / "frontend" / "node_modules"),
                     "STOPROW_FRONT": "http://127.0.0.1:%d" % front,
                     "STOPROW_OUT": str(dest), "STOPROW_SHOT": str(shot), "STOPROW_JID": jid})
        node = subprocess.run(["node", str(PROBE)], capture_output=True,
                              text=True, encoding="utf-8", errors="replace", env=penv, timeout=420)
        out["probe_rc"] = node.returncode
        out["probe_stdout"] = (node.stdout or "")[-400:]
        out["probe_stderr"] = (node.stderr or "")[-400:]
        rep = json.loads(dest.read_text(encoding="utf-8")) if dest.exists() else {}
        # 形制闸：回执缺哪一腿的键，那一腿就只能写 NOT_MEASURED，绝不许把"空列表"读成"没有行被压住"。
        want = ("static_frame", "if_needed_rows", "above_bar_rows", "bar_band", "counts")
        missing = [k for k in want if k not in rep]
        out["probe_schema_missing_keys"] = missing
        if missing:
            BAD.append("探针回执缺键 %s —— 这台尺的对应腿没有读数（%s）"
                       % (",".join(missing), PROBE.name))
        sf = rep.get("static_frame") or {}
        rows = sf.get("rows") or []
        ifrows = rep.get("if_needed_rows") or []
        abrows = rep.get("above_bar_rows") or []
        legs = [{k: v for k, v in l.items() if k != "inventory"} for l in (rep.get("legs") or [])]
        inv = {l["leg"]: {k: v["count"] for k, v in l["inventory"].items()}
               for l in (rep.get("legs") or [])}
        out["screen"] = {"legs": legs, "anchor_counts": inv,
                         "anchors_requested": rep.get("anchors_requested"),
                         "frame": sf.get("frame"), "fixed_elements": sf.get("fixed_elements"),
                         "bar_band": rep.get("bar_band"), "counts": rep.get("counts"),
                         "static_rows": rows, "if_needed_rows": ifrows, "above_bar_rows": abrows,
                         "errors": rep.get("errors"), "viewport": rep.get("viewport"),
                         "screenshot": shot.name}
        fixture_text = fx.read_text(encoding="utf-8", errors="replace")
        out["name_triple_check"] = {
            "api_titles": out["api_stop_titles"], "screen_rows": len(rows),
            "every_static_row_has_a_rect": bool(rows) and all(r.get("rect") for r in rows),
            "every_above_bar_row_hits": bool(abrows) and all(
                (r.get("top_of_stack") or {}).get("tag") for r in abrows),
            "every_above_bar_row_clears_bar": bool(abrows) and all(
                bool(r.get("clears_bar")) for r in abrows),
            "under_bar_in_leg_B": [r["index"] for r in ifrows if r.get("under_bar")],
            "center_outside_viewport_in_leg_A": [r["index"] for r in rows
                                                  if not r.get("center_in_viewport")],
            "per_name": [{"title": t, "in_fixture_file": t in fixture_text,
                          "on_screen": any(t and t in (r.get("text") or "") for r in rows)}
                         for t in out["api_stop_titles"]],
        }
        if not rows:
            BAD.append("屏上一枚停靠行都没读到（探针回执 %s）" % dest.name)
        elif not all(x["in_fixture_file"] and x["on_screen"]
                     for x in out["name_triple_check"]["per_name"]):
            BAD.append("三处名字对不齐，块里不能写「同名三处对上」")
    except SystemExit:
        pass
    finally:
        srv.shutdown()
        srv.server_close()
        backend.terminate()
        try:
            backend.wait(timeout=40)
        except subprocess.TimeoutExpired:
            backend.kill()
        out["backend_exit"] = backend.poll()
    out["BAD"] = BAD
    dst = ART / ("stoprow_walk_%s.json" % run_id)
    dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + chr(10), encoding="utf-8",
                   newline=chr(10))
    print(json.dumps({k: out[k] for k in ("run_id", "probe", "driver", "line_tip_at_walk", "build_tip",
                                          "api_port", "front_port", "ensure", "api_stop_count",
                                          "byte_pairing", "fixture", "probe_rc", "probe_stdout")
                      if k in out}, ensure_ascii=False, indent=2))
    tri = out.get("name_triple_check", {})
    sc = out.get("screen", {})
    print("屏读：静态行 %s 枚｜都有矩形 %s｜B 半被底栏压住 %s｜C 半都在底栏之上 %s｜C 半都命中 %s"
          % (tri.get("screen_rows"), tri.get("every_static_row_has_a_rect"),
             tri.get("under_bar_in_leg_B"), tri.get("every_above_bar_row_clears_bar"),
             tri.get("every_above_bar_row_hits")))
    print("帧", json.dumps(sc.get("frame")), "| 底栏", json.dumps(
        {"count": (sc.get("bar_band") or {}).get("count"),
         "top": (sc.get("bar_band") or {}).get("top"),
         "cls": [(x.get("cls"), x.get("rect"), x.get("z"))
                 for x in ((sc.get("bar_band") or {}).get("items") or [])]}, ensure_ascii=False))
    for r in (sc.get("static_rows") or [])[:10]:
        b = r.get("rect") or {}
        print("   S#%s 文档帧 y=%s bottom=%s h=%s w=%s 中心在视口=%s 字=%r"
              % (r.get("index"), b.get("y"), b.get("bottom"), b.get("height"), b.get("width"),
                 r.get("center_in_viewport"), (r.get("text") or "")[:56]))
    for r in (sc.get("if_needed_rows") or [])[:10]:
        b = r.get("rect") or {}
        print("   B#%s scroll_y=%s y=%s bottom=%s 压底栏=%s 命中=%s"
              % (r.get("index"), r.get("scroll_y"), b.get("y"), b.get("bottom"),
                 r.get("under_bar"), json.dumps(r.get("top_of_stack"), ensure_ascii=False)))
    for r in (sc.get("above_bar_rows") or [])[:10]:
        b = r.get("rect") or {}
        print("   C#%s scroll_y=%s y=%s bottom=%s 越过底栏=%s 命中=%s 叠=%s"
              % (r.get("index"), r.get("scroll_y"), b.get("y"), b.get("bottom"),
                 r.get("clears_bar"), json.dumps(r.get("top_of_stack"), ensure_ascii=False),
                 json.dumps(r.get("stack"), ensure_ascii=False)[:150]))
    print("锚计数", json.dumps(sc.get("anchor_counts", {}), ensure_ascii=False))
    print("ARTIFACT", dst)
    print("BAD", BAD)
    return 1 if BAD else 0


if __name__ == "__main__":
    sys.exit(main())
