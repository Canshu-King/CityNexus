def g(*a: str) -> str:

def http(url, method="GET", headers=None, body=None):
    r = urllib.request.Request(url, data=body, method=method, headers=headers or {})
    try:
        with urllib.request.urlopen(r, timeout=60) as resp:
            return resp.status, resp.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()
    except Exception as e:  # noqa: BLE001
        return None, (type(e).__name__ + " " + str(e))[:120].encode()


def all_counts(db):
    with sqlite3.connect("file:%s?mode=ro" % db.as_posix(), uri=True) as con:
        names = [r[0] for r in con.execute(
            "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'"
        ).fetchall()]
        out = {}
        for n in sorted(names):
            out[n] = con.execute('SELECT count(*) FROM "%s"' % n).fetchone()[0]
    return out


def journey_census(db, jid):
    """每一张带 journey_id 列的表：该行程下几行 + 列名 + 头一行的形状摘要。"""

    with sqlite3.connect("file:%s?mode=ro" % db.as_posix(), uri=True) as con:
        names = [r[0] for r in con.execute(
            "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'"
        ).fetchall()]
        per = {}
        for n in sorted(names):
            cols = [r[1] for r in con.execute('PRAGMA table_info("%s")' % n).fetchall()]
            if "journey_id" not in cols:
                continue
            cnt = con.execute('SELECT count(*) FROM "%s" WHERE journey_id=?' % n,
                              (jid,)).fetchone()[0]
            sample = None
            if cnt:
                row = con.execute('SELECT * FROM "%s" WHERE journey_id=? LIMIT 1' % n,
                                  (jid,)).fetchone()
                sample = {}
                for c, v in zip(cols, row):
                    s = str(v)
                    sample[c] = (s[:36] + "…") if len(s) > 37 else v
            per[n] = {"columns": cols, "rows_for_this_journey": cnt, "first_row": sample}
        # journeys 表自身那一行
        jrow = con.execute("SELECT id, status, city_code, active_plan_version_id,"
                           " current_activity_id, start_at, end_at FROM journeys WHERE id=?",
                           (jid,)).fetchone()
    return {"tables_with_journey_id_fk": per,
            "journey_self": None if not jrow else {
                "id": str(jrow[0]), "status": jrow[1], "city_code": jrow[2],
                "active_plan_version_id": str(jrow[3]) if jrow[3] else None,
                "current_activity_id": str(jrow[4]) if jrow[4] else None,
                "start_at_set": jrow[5] is not None, "end_at_set": jrow[6] is not None}}


def model_tablenames(tip):
    """把 `models.py` 里的类名与 __tablename__ 抽出来，好让"哪张表"能对上产品自己的名字。"""

    src = g("show", "%s:backend/app/db/models.py" % tip)
    out = {}
    cls = None
    for line in src.splitlines():
        m = re.match(r"^class (\w+)", line)
        if m:
            cls = m.group(1)
            continue
        m = re.match(r'^\s+__tablename__\s*=\s*["\'](\w+)["\']', line)
        if m and cls:
            out[cls] = m.group(1)
    return out


def main() -> int:

    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)

                          capture_output=True)

        return 1

    tar.write_bytes(proc.stdout)

    with tarfile.open(str(tar)) as t:
        t.extractall(path=str(snap))
    tar.unlink()
    blobs = {}
    for rel in CHECKED:
        want, have = g("rev-parse", "%s:%s" % (tip, rel)), g("hash-object", "--no-filters",
                                                             str(snap / rel))
        if len(want) != 40 or have != want:
            print("REFUSED: 导出字节与尖不等", rel)
            return 1
        blobs[rel] = want

    db = snap / "backend" / ("%s.db" % run_id)
    port = mod.free_port(int(os.environ.get("DAY25_PORT", "4801")), "backend")
    env = mod.env_base(run_id, db, port)
    env["CITYNEXUS_DEMO_MODE"] = "true"
    token = secrets.token_hex(16)
    env["CITYNEXUS_DEMO_SCENE_TOKEN"] = token
    base = "http://127.0.0.1:%d" % port
    hdr = {"Content-Type": "application/json", "X-Demo-Scene-Token": token,
           "Idempotency-Key": "day25-" + run_id}
    out = {"run_id": run_id, "line_tip": tip, "blobs": blobs, "backend_port": port,
           "token_env_name": "CITYNEXUS_DEMO_SCENE_TOKEN",
           "token_header_name": "X-Demo-Scene-Token"}
    child = subprocess.Popen([mod.PY, "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1",
                              "--port", str(port)], cwd=str(snap / "backend"), env=env,
                             stdout=(ART / ("uvicorn_%s.log" % run_id)).open("w", encoding="utf-8"),
                             stderr=subprocess.STDOUT)
    out["child_pid"] = child.pid
    try:
        ok, reading = mod.wait_http(base + "/api/v1/health")
        hs, hb = http(base + "/api/v1/health")
        out["health"] = {"ok": ok, "http_status": hs,
                         "carries_this_run_id": run_id in hb.decode("utf-8", "replace")}
        if not out["health"]["carries_this_run_id"]:
            BAD.append("归属未由本 run id 证")
        out["counts_before"] = all_counts(db)
        st, body = http(base + "/api/v1/demo/scene")
        out["scene_before"] = {"http_status": st, "json": json.loads(body.decode("utf-8"))}
        st, body = http(base + "/api/v1/demo/scene/ensure", "POST", hdr, b"{}")
        receipt = json.loads(body.decode("utf-8"))
        out["ensure"] = {"http_status": st, "state": receipt.get("state"),
                         "journey_id": receipt.get("journey_id"),
                         "rows_installed": (receipt.get("supply") or {}).get("rows_installed"),
                         "rows_deleted": (receipt.get("supply") or {}).get("rows_deleted"),
                         "destructive": (receipt.get("supply") or {}).get("destructive"),
                         "advance_steps": [(s.get("step"), s.get("ok"))
                                           for s in (receipt.get("advance") or {}).get("steps") or []],
                         "journey_status": receipt.get("journey_status"),
                         "scene_label": receipt.get("scene_label")}
        if st != 200 or receipt.get("state") != "READY":
            BAD.append("新尖上空库首供不是 200 READY —— 块里的话要改")
        out["counts_after"] = all_counts(db)
        jid = receipt.get("journey_id")
        out["table_diff"] = {k: [out["counts_before"].get(k), v]
                             for k, v in out["counts_after"].items()
                             if out["counts_before"].get(k) != v}
        out["journey_census"] = journey_census(db, jid)
        st, body = http(base + "/api/v1/demo/scene")
        out["scene_after"] = {"http_status": st, "json": json.loads(body.decode("utf-8"))}
        if out["scene_after"]["json"].get("journey_id") != jid:
            BAD.append("供完之后 GET /demo/scene 没点名同一枚行程")
        out["model_tablenames"] = model_tablenames(tip)
        for path in ("/api/v1/journeys/%s" % jid, "/api/v1/journeys/%s/itinerary" % jid):
            s2, b2 = http(base + path)
            try:
                j = json.loads(b2.decode("utf-8"))
            except Exception:  # noqa: BLE001
                j = None
            out.setdefault("read_paths", []).append(
                {"path": path.replace(jid, "<journey_id>"), "http_status": s2,
                 "bytes": len(b2),
                 "top_keys": sorted(j.keys())[:12] if isinstance(j, dict) else
                            ("LIST len %d" % len(j) if isinstance(j, list) else "NOT_JSON"),
                 "items": (len(j) if isinstance(j, list) else
                           len(j.get("items", [])) if isinstance(j, dict)
                           and isinstance(j.get("items"), list) else None)})
    finally:
        child.terminate()
        try:
            child.wait(timeout=60)
        except subprocess.TimeoutExpired:
            child.kill()
        out["child_exit"] = child.poll()
    out["BAD"] = BAD
    dst = ART / ("sealed_demo_day25_census_%s.json" % run_id)
    dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + "\n", encoding="utf-8",
                   newline=chr(10))
    print(json.dumps({k: out[k] for k in ("line_tip", "ensure", "table_diff", "read_paths")},
                     ensure_ascii=False, indent=2)[:2600])
    print("CENSUS tables with journey_id:",
          {t: v["rows_for_this_journey"] for t, v in
           out["journey_census"]["tables_with_journey_id_fk"].items()})
    print("ARTIFACT", dst, "| BAD", BAD)
    return 1 if BAD else 0


if __name__ == "__main__":
    sys.exit(main())
