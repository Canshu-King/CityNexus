# -*- coding: utf-8 -*-
"""day25 上的「手机那一扇真正读到什么」：全新库、一次 ensure，然后把读侧答复**全文**存进件，
并把表级链（安排在哪张表、站点身份在哪张表、journey_legs 到底几行）在同一趟里再数一遍。
复用上一趟已验过字节的导出目录（本趟开头再逐枚回读一次哈希），token 只活在环境里。"""
from __future__ import annotations

import glob
import importlib.util
import json
import os
import secrets
import sqlite3
import subprocess
import sys
import time
import urllib.error
import urllib.request
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
RUNNER = WT2 / "review/ord0502/run_ord0502_stack_r1.py"
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
CHECKED = ("backend/app/main.py", "backend/app/demo_scene.py", "backend/app/db/models.py")
BAD: list[str] = []


def g(*a: str) -> str:
    return (subprocess.run(["git", *a], cwd=str(WT2), capture_output=True, text=True,
                           encoding="utf-8", errors="replace").stdout or "").strip()


def http(url, headers=None, body=None, method=None):
    m = method or ("POST" if body is not None else "GET")
    r = urllib.request.Request(url, data=body, method=m, headers=headers or {})
    try:
        with urllib.request.urlopen(r, timeout=60) as resp:
            return resp.status, resp.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()


def main() -> int:
    spec = importlib.util.spec_from_file_location("ord0502_runner", str(RUNNER))
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    tip = g("ls-remote", "--refs", "origin", MIRROR).split("\t")[0]
    if len(tip) != 40:
        print("REFUSED: 读不到尖")
        return 1
    prev = sorted(glob.glob(str(Path("D:/moodle") / "_qoder_day25_*")))
    if not prev:
        print("REFUSED: 上一趟的导出目录已不在，需先重导出")
        return 1
    snap = Path(prev[-1])
    blobs = {}
    for rel in CHECKED:
        want, have = g("rev-parse", "%s:%s" % (tip, rel)), g("hash-object", "--no-filters",
                                                             str(snap / rel))
        if len(want) != 40 or have != want:
            print("REFUSED: 这枚导出的字节不等于新尖上的那枚", rel)
            return 1
        blobs[rel] = want

    run_id = "DAY25LV-" + time.strftime("%H%M%S")
    db = snap / "backend" / ("%s.db" % run_id)
    if db.exists():
        print("REFUSED: 库必须是不存在的新文件")
        return 1
    port = mod.free_port(int(os.environ.get("DAY25_PORT", "4803")), "backend")
    env = mod.env_base(run_id, db, port)
    env["CITYNEXUS_DEMO_MODE"] = "true"
    token = secrets.token_hex(16)
    env["CITYNEXUS_DEMO_SCENE_TOKEN"] = token
    base = "http://127.0.0.1:%d" % port
    hdr = {"Content-Type": "application/json", "X-Demo-Scene-Token": token,
           "Idempotency-Key": "day25lv-" + run_id}
    out = {"run_id": run_id, "line_tip": tip, "blobs": blobs, "snap": str(snap),
           "backend_port": port, "db": db.as_posix()}
    child = subprocess.Popen([mod.PY, "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1",
                              "--port", str(port)], cwd=str(snap / "backend"), env=env,
                             stdout=(ART / ("uvicorn_%s.log" % run_id)).open("w", encoding="utf-8"),
                             stderr=subprocess.STDOUT)
    out["child_pid"] = child.pid
    try:
        ok, reading = mod.wait_http(base + "/api/v1/health")
        hs, hb = http(base + "/api/v1/health")
        out["health"] = {"ok": ok, "http_status": hs,
                         "carries_this_run_id": run_id in hb.decode("utf-8", "replace")}
        if not out["health"]["carries_this_run_id"]:
            BAD.append("归属未证")
        st, body = http(base + "/api/v1/demo/scene/ensure", hdr, body=b"{}", method="POST")
        rec = json.loads(body.decode("utf-8"))
        jid = rec.get("journey_id")
        out["ensure"] = {"http_status": st, "state": rec.get("state"), "journey_id": jid,
                         "journey_status": rec.get("journey_status"),
                         "rows_installed": (rec.get("supply") or {}).get("rows_installed")}
        if st != 200 or rec.get("state") != "READY":
            BAD.append("新库首供不是 200 READY")
        if not jid:
            out["ensure_raw_head"] = body[:400].decode("utf-8", "replace")
            out["ensure_raw_keys"] = sorted(rec.keys()) if isinstance(rec, dict) else None
            BAD.append("首供答复里没有行程号 —— 读侧那几腿不成立，先把答复原文交出来")
            print(json.dumps(out, ensure_ascii=False, indent=2)[:1500])
            dst = ART / ("day25_read_side_failure_%s.json" % run_id)
            dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + "\n",
                           encoding="utf-8", newline=chr(10))
            print("ARTIFACT", dst)
            return 1
        # 读侧全文
        reads = {}
        for name, path in (("journeys_by_id", "/api/v1/journeys/%s" % jid),
                           ("demo_scene", "/api/v1/demo/scene"),
                           ("journey_plans", "/api/v1/journeys/%s/plans" % jid),
                           ("journey_calendar", "/api/v1/journeys/%s/calendar" % jid),
                           ("journey_activities", "/api/v1/journeys/%s/activities" % jid)):
            s2, b2 = http(base + path)
            parsed = None
            try:
                parsed = json.loads(b2.decode("utf-8"))
            except Exception:  # noqa: BLE001
                parsed = None
            reads[name] = {"path": path.replace(jid, "<journey_id>"), "http_status": s2,
                           "bytes": len(b2),
                           "json": parsed if isinstance(parsed, (dict, list)) else None,
                           "non_json_head": None if parsed is not None else b2[:120].decode(
                               "utf-8", "replace")}
        out["read_side"] = reads
        lv = (((reads.get("journeys_by_id") or {}).get("json") or {}).get("live_view"))
        out["live_view_type"] = type(lv).__name__
        out["live_view_keys"] = sorted(lv.keys())[:24] if isinstance(lv, dict) else None
        out["live_view_stored_in_full"] = lv if isinstance(lv, dict) else None
        # 表级链再数一遍（同一趟，避免跨趟外推）
        con = sqlite3.connect("file:%s?mode=ro" % db.as_posix(), uri=True)
        pv = con.execute("SELECT active_plan_version_id FROM journeys WHERE id=?",
                         (jid,)).fetchone()[0]
        out["chain"] = {
            "active_plan_version_id": str(pv),
            "plan_versions_for_this_journey": con.execute(
                "SELECT count(*) FROM plan_versions WHERE journey_id=?", (jid,)).fetchone()[0],
            "activities_total": con.execute("SELECT count(*) FROM activities").fetchone()[0],
            "activities_on_active_plan_version": con.execute(
                "SELECT count(*) FROM activities WHERE plan_version_id=?", (str(pv),)).fetchone()[0],
            "activities_with_place_id_on_active": con.execute(
                "SELECT count(*) FROM activities WHERE plan_version_id=? AND place_id IS NOT NULL",
                (str(pv),)).fetchone()[0],
            "distinct_place_ids": con.execute(
                "SELECT count(DISTINCT place_id) FROM activities").fetchone()[0],
            "places_total": con.execute("SELECT count(*) FROM places").fetchone()[0],
            "journey_legs_total": con.execute("SELECT count(*) FROM journey_legs").fetchone()[0],
            "sequence_and_titles": [
                [r[0], r[1], r[2], r[3]] for r in con.execute(
                    "SELECT sequence_no, title, place_id, activity_type FROM activities "
                    "WHERE plan_version_id=? ORDER BY sequence_no, id", (str(pv),)).fetchall()],
        }
        con.close()
    finally:
        child.terminate()
        try:
            child.wait(timeout=60)
        except subprocess.TimeoutExpired:
            child.kill()
        out["child_exit"] = child.poll()
    out["BAD"] = BAD
    dst = ART / ("day25_read_side_%s.json" % run_id)
    dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + "\n", encoding="utf-8",
                   newline=chr(10))
    print(json.dumps({"ensure": out["ensure"], "chain": out["chain"],
                      "live_view_keys": out["live_view_keys"],
                      "read_status": {k: v["http_status"] for k, v in out["read_side"].items()}},
                     ensure_ascii=False, indent=2)[:2400])
    print("ARTIFACT", dst, "| BAD", BAD)
    return 1 if BAD else 0


if __name__ == "__main__":
    sys.exit(main())
