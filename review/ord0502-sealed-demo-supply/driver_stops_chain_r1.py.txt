# -*- coding: utf-8 -*-
"""只读普查：供完之后，「站点」与「安排」到底各落在哪张表。
不重起进程——直接以 mode=ro 打开本 run 那枚私有库，按列名找链，再与产品自己声明的
模型/路由对一遍（`git show <尖>:…` 现读），不猜表名。"""
from __future__ import annotations

import glob
import json
import re
import sqlite3
import subprocess
import sys
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8", errors="replace")

WT2 = Path("D:/moodle/_qoder_newlane_ord0444_20260929/wt2")
ART = Path("D:/moodle/_qoder_anchor_rulers_20261002")
MIRROR = "refs/heads/mirror/integration-frontend-apple-pass1"
CENSUS = sorted(glob.glob(str(ART / "sealed_demo_day25_census_*.json")))[-1]


def g(*a: str) -> str:
    return (subprocess.run(["git", *a], cwd=str(WT2), capture_output=True, text=True,
                           encoding="utf-8", errors="replace").stdout or "").strip()


def main() -> int:
    d = json.loads(Path(CENSUS).read_text(encoding="utf-8"))
    tip, jid, pv = d["line_tip"], d["ensure"]["journey_id"], \
        d["journey_census"]["journey_self"]["active_plan_version_id"]
    pattern = str(Path("D:/moodle") / ("_qoder_day25_%s" % d["run_id"]) / "backend" / "*.db")
    dbs = glob.glob(pattern)
    if not dbs:
        print("REFUSED: 找不到本 run 那枚库", pattern)
        return 1
    db = Path(dbs[0])
    con = sqlite3.connect("file:%s?mode=ro" % db.as_posix(), uri=True)
    tables = [r[0] for r in con.execute(
        "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'"
    ).fetchall()]
    out = {"run_id": d["run_id"], "line_tip": tip, "db": db.as_posix(),
           "journey_id": jid, "active_plan_version_id": pv, "tables": {}}

    def cols(t):
        return [r[1] for r in con.execute('PRAGMA table_info("%s")' % t).fetchall()]

    for t in sorted(tables):
        c = cols(t)
        hit = {}
        if "journey_id" in c:
            hit["by_journey_id"] = con.execute(
                'SELECT count(*) FROM "%s" WHERE journey_id=?' % t, (jid,)).fetchone()[0]
        if "plan_version_id" in c:
            hit["by_active_plan_version"] = con.execute(
                'SELECT count(*) FROM "%s" WHERE plan_version_id=?' % t, (pv,)).fetchone()[0]
        if "place_id" in c:
            hit["rows_with_place_id"] = con.execute(
                'SELECT count(*) FROM "%s" WHERE place_id IS NOT NULL' % t).fetchone()[0]
        if hit:
            hit["columns"] = c
            hit["total_rows"] = con.execute('SELECT count(*) FROM "%s"' % t).fetchone()[0]
            out["tables"][t] = hit

    # 那一枚活动行长什么样（安排的具体形状），以及它指向哪枚站点身份
    acts = out["tables"].get("activities", {})
    if acts:
        rows = con.execute('SELECT * FROM activities WHERE plan_version_id=? '
                           'ORDER BY rowid LIMIT 3', (pv,)).fetchall()
        out["activities_sample"] = [dict(zip(acts["columns"], r)) for r in rows]
        out["activities_place_kinds"] = con.execute(
            'SELECT place_id, count(*) FROM activities WHERE plan_version_id=? '
            'GROUP BY place_id', (pv,)).fetchall().__len__()
        ids = [r[0] for r in con.execute(
            'SELECT DISTINCT place_id FROM activities WHERE place_id IS NOT NULL').fetchall()]
        out["distinct_place_ids"] = len(ids)
        out["place_ids_in_catalog"] = con.execute(
            'SELECT count(*) FROM places WHERE id IN (%s)' % ",".join("?" * len(ids)),
            ids).fetchone()[0] if ids else 0
    # 产品自己怎么说这两张表：模型与建表处 + 行程屏读的路由
    src = g("show", "%s:backend/app/db/models.py" % tip)
    names = {}
    cls = None
    for line in src.splitlines():
        m = re.match(r"^class (\w+)", line)
        if m:
            cls = m.group(1)
        m = re.match(r'^\s+__tablename__\s*=\s*["\'](\w+)["\']', line)
        if m and cls:
            names[cls] = m.group(1)
    out["model_names"] = {k: names[k] for k in ("Activity", "Place", "PlanVersion", "JourneyLeg")
                          if k in names}
    leg_hits = [l for l in src.splitlines() if re.match(r"^class (Activity|JourneyLeg|Place)\b", l)]
    out["model_lines"] = {}
    for name in ("Activity", "JourneyLeg", "Place"):
        found = g("grep", "-n", "-e", "class %s(" % name, tip, "--", "backend/app/db/models.py")
        first = found.split("\n")[0] if found else ""
        m = re.match(r"^[^:]+:(\d+):", first)
        if m:
            out["model_lines"][name] = "backend/app/db/models.py:%s" % m.group(1)
    if len(out["model_lines"]) < 3:
        out["model_lines"]["_note"] = "有一枚类名在尖上没按 'class X(' 命中，具名 NOT_LOCATED"
    legs_route = g("grep", "-n", "-e", "journey_legs", tip, "--", "backend/app")
    out["who_touches_journey_legs"] = legs_route[:1200]
    con.close()
    dst = ART / ("stops_schedule_chain_%s.json" % d["run_id"])
    dst.write_text(json.dumps(out, ensure_ascii=False, indent=2) + "\n", encoding="utf-8",
                   newline=chr(10))
    print(json.dumps({"tables_with_rows_for_this_journey": {
        t: v for t, v in out["tables"].items()
        if v.get("by_journey_id") or v.get("by_active_plan_version")}},
        ensure_ascii=False, indent=1)[:2000])
    print("model_names", out["model_names"], "| model_lines", out["model_lines"])
    print("activities", json.dumps({k: out["tables"]["activities"][k] for k in
                                   ("total_rows", "by_active_plan_version", "columns")},
                                   ensure_ascii=False)[:600] if "activities" in out["tables"] else "-")
    print("distinct_place_ids", out.get("distinct_place_ids"),
          "in_catalog", out.get("place_ids_in_catalog"), "| journey_legs rows",
          out["tables"].get("journey_legs", {}).get("total_rows"))
    print("ARTIFACT", dst)
    return 0


if __name__ == "__main__":
    sys.exit(main())
